   131→   131→function atrSeries(bars: ScalpBar[], len: number): number[] {
   132→   132→  const out: number[] = new Array(bars.length).fill(NaN);
   133→   133→  if (bars.length < len + 1) return out;
   134→   134→  const trs: number[] = [bars[0].h - bars[0].l];
   135→   135→  for (let i = 1; i < bars.length; i++) {
   136→   136→    trs.push(
   137→   137→      Math.max(
   138→   138→        bars[i].h - bars[i].l,
   139→   139→        Math.abs(bars[i].h - bars[i - 1].c),
   140→   140→        Math.abs(bars[i].l - bars[i - 1].c),
   141→   141→      ),
   142→   142→    );
   143→   143→  }
   144→   144→  let a = 0;
   145→   145→  for (let i = 0; i < len; i++) a += trs[i];
   146→   146→  out[len] = a / len;
   147→   147→  for (let i = len + 1; i < bars.length; i++) {
   148→   148→    a = (a * (len - 1) + trs[i]) / len;
   149→   149→    out[i] = a;
   150→   150→  }
   151→   151→  return out;
   152→   152→}

   154→   154→function rma(values: number[], len: number): number[] {
   155→   155→  const out: number[] = new Array(values.length).fill(NaN);
   156→   156→  let sum = 0;
   157→   157→  let prev = NaN;
   158→   158→  for (let i = 0; i < values.length; i++) {
   159→   159→    if (i < len) {
   160→   160→      sum += values[i];
   161→   161→      if (i === len - 1) {
   162→   162→        prev = sum / len;
   163→   163→        out[i] = prev;
   164→   164→      }
   165→   165→    } else {
   166→   166→      prev = (prev * (len - 1) + values[i]) / len;
   167→   167→      out[i] = prev;
   168→   168→    }
   169→   169→  }
   170→   170→  return out;
   171→   171→}

   173→   173→function rsiSeries(c: number[], len = 14): number[] {
   174→   174→  const ch: number[] = [0];
   175→   175→  for (let i = 1; i < c.length; i++) ch.push(c[i] - c[i - 1]);
   176→   176→  const gains = ch.map((d) => (d > 0 ? d : 0));
   177→   177→  const losses = ch.map((d) => (d < 0 ? -d : 0));
   178→   178→  const ag = rma(gains, len);
   179→   179→  const al = rma(losses, len);
   180→   180→  const out: number[] = c.map(() => NaN);
   181→   181→  for (let i = 0; i < c.length; i++) {
   182→   182→    if (isNaN(ag[i]) || isNaN(al[i])) continue;
   183→   183→    out[i] = al[i] === 0 ? (ag[i] === 0 ? 50 : 100) : 100 - 100 / (1 + ag[i] / al[i]);
   184→   184→  }
   185→   185→  return out;
   186→   186→}

   188→   188→function macdHistSeries(c: number[], fast = 12, slow = 26, sig = 9): number[] {
   189→   189→  const ema = (len: number): number[] => {
   190→   190→    const k = 2 / (len + 1);
   191→   191→    const out: number[] = [];
   192→   192→    let e = c[0];
   193→   193→    for (let i = 0; i < c.length; i++) {
   194→   194→      e = i === 0 ? c[0] : c[i] * k + e * (1 - k);
   195→   195→      out.push(e);
   196→   196→    }
   197→   197→    return out;
   198→   198→  };
   199→   199→  const f = ema(fast);
   200→   200→  const s = ema(slow);
   201→   201→  const line = f.map((v, i) => v - s[i]);
   202→   202→  const k = 2 / (sig + 1);
   203→   203→  let se = line[0];
   204→   204→  return line.map((v, i) => {
   205→   205→    se = i === 0 ? v : v * k + se * (1 - k);
   206→   206→    return v - se;
   207→   207→  });
   208→   208→}

   210→   210→function stochSeries(bars: ScalpBar[], len = 14): number[] {
   211→   211→  const out: number[] = bars.map(() => NaN);
   212→   212→  for (let i = len - 1; i < bars.length; i++) {
   213→   213→    let hh = -Infinity;
   214→   214→    let ll = Infinity;
   215→   215→    for (let j = i - len + 1; j <= i; j++) {
   216→   216→      if (bars[j].h > hh) hh = bars[j].h;
   217→   217→      if (bars[j].l < ll) ll = bars[j].l;
   218→   218→    }
   219→   219→    out[i] = hh === ll ? 50 : ((bars[i].c - ll) / (hh - ll)) * 100;
   220→   220→  }
   221→   221→  return out;
   222→   222→}

   224→   224→function cciSeries(bars: ScalpBar[], len = 20): number[] {
   225→   225→  const tp = bars.map((b) => (b.h + b.l + b.c) / 3);
   226→   226→  const out: number[] = bars.map(() => NaN);
   227→   227→  for (let i = len - 1; i < bars.length; i++) {
   228→   228→    let mean = 0;
   229→   229→    for (let j = i - len + 1; j <= i; j++) mean += tp[j];
   230→   230→    mean /= len;
   231→   231→    let dev = 0;
   232→   232→    for (let j = i - len + 1; j <= i; j++) dev += Math.abs(tp[j] - mean);
   233→   233→    dev /= len;
   234→   234→    out[i] = dev === 0 ? 0 : (tp[i] - mean) / (0.015 * dev);
   235→   235→  }
   236→   236→  return out;
   237→   237→}

   239→   239→function mfiSeries(bars: ScalpBar[], len = 14): number[] {
   240→   240→  const tp = bars.map((b) => (b.h + b.l + b.c) / 3);
   241→   241→  const out: number[] = bars.map(() => NaN);
   242→   242→  for (let i = len; i < bars.length; i++) {
   243→   243→    let pos = 0;
   244→   244→    let neg = 0;
   245→   245→    for (let j = i - len + 1; j <= i; j++) {
   246→   246→      const flow = tp[j] * bars[j].c;
   247→   247→      if (tp[j] > tp[j - 1]) pos += flow;
   248→   248→      else if (tp[j] < tp[j - 1]) neg += flow;
   249→   249→    }
   250→   250→    out[i] = neg === 0 ? 100 : pos === 0 ? 0 : 100 - 100 / (1 + pos / neg);
   251→   251→  }
   252→   252→  return out;
   253→   253→}

   255→   255→interface Pivot {
   256→   256→  idx: number; // اندیس کندل پیوت
   257→   257→  price: number;
   258→   258→}
   259→   259→
   260→   260→/** پیوت‌های تأییدشده — پیوت در کندل idx وقتی تأیید می‌شود که idx+right پردازش شود */
   261→   261→function pivots(highs: number[], lows: number[], left: number, right: number) {
   262→   262→  const highsP: Pivot[] = [];
   263→   263→  const lowsP: Pivot[] = [];
   264→   264→  for (let i = left; i < highs.length - right; i++) {
   265→   265→    let isH = true;
   266→   266→    let isL = true;
   267→   267→    for (let k = 1; k <= left; k++) {
   268→   268→      if (highs[i - k] >= highs[i]) isH = false;
   269→   269→      if (lows[i - k] <= lows[i]) isL = false;
   270→   270→    }
   271→   271→    for (let k = 1; k <= right; k++) {
   272→   272→      if (highs[i + k] >= highs[i]) isH = false;
   273→   273→      if (lows[i + k] <= lows[i]) isL = false;
   274→   274→    }
   275→   275→    if (isH) highsP.push({ idx: i, price: highs[i] });
   276→   276→    if (isL) lowsP.push({ idx: i, price: lows[i] });
   277→   277→  }
   278→   278→  return { highsP, lowsP };
   279→   279→}

   281→   281→function rp(v: number): number {
   282→   282→  if (!isFinite(v)) return 0;
   283→   283→  if (v >= 1000) return Math.round(v * 100) / 100;
   284→   284→  if (v >= 10) return Math.round(v * 100) / 100;
   285→   285→  return Math.round(v * 100000) / 100000;
   286→   286→}
