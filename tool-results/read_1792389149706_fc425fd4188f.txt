    84→  try {
    85→    navigator.vibrate?.(12);
    86→  } catch {
    87→    /* پشتیبانی نمی‌شود */
    88→  }

   100→/* ---------- پوسته ---------- */
   101→
   102→function Shell() {
   103→  const { ms, msRefreshing, refreshMarket, now } = useShell();
   104→
   105→  /* حالت لرزش و جابه‌جایی آیکون‌ها */
   106→  const [wiggle, setWiggle] = useState(false);
   107→  const [gridOrder, setGridOrder] = useState<AppId[]>(DEFAULT_GRID_ORDER);
   108→  const [drag, setDrag] = useState<{ id: AppId; x: number; y: number } | null>(null);
   109→
   110→  /* مراکز اعلان و کنترل */
   111→  const [ncOpen, setNcOpen] = useState(false);
   112→  const [ccOpen, setCcOpen] = useState(false);
   113→  const [ncProgress, setNcProgress] = useState(0);
   114→  const [ccProgress, setCcProgress] = useState(0);
   115→  const ncProgRef = useRef(0);
   116→  const ccProgRef = useRef(0);
   117→
   118→  /* منوی جانبی */
   119→  const [drawerOpen, setDrawerOpen] = useState(false);
   120→
   121→  const setNc = useCallback((v: number) => {
   122→    ncProgRef.current = v;
   123→    setNcProgress(v);
   124→  }, []);
   125→  const setCc = useCallback((v: number) => {
   126→    ccProgRef.current = v;
   127→    setCcProgress(v);
   128→  }, []);
   129→
   130→  /* اجرای اپ با انیمیشن بزرگ‌نمایی از موقعیت آیکون (FLIP) */
   131→  const [app, setApp] = useState<{ id: AppId; rect: AppRect } | null>(null);
   132→  const [appPhase, setAppPhase] = useState<"opening" | "open" | "closing">("open");
   133→
   134→  /* ---------- لایه‌های تاریخچه — دکمه برگشت خود گوشی ---------- */
   135→  const lastPushedRef = useRef<string | null>(null);
   136→  const pushLayer = useCallback((name: string) => {
   137→    lastPushedRef.current = name;
   138→    try {
   139→      history.pushState({ sd: name }, "");
   140→    } catch {
   141→      lastPushedRef.current = null;
   142→    }
   143→  }, []);
   144→
   145→  const closeApp = useCallback(() => {
   146→    setAppPhase("closing");
   147→    setTimeout(() => setApp(null), 420);
   148→  }, []);
   149→
   150→  const closeLayerUI = useCallback(
   151→    (name: "app" | "nc" | "cc" | "drawer") => {
   152→      if (lastPushedRef.current === name) {
   153→        lastPushedRef.current = null;
   154→        try {
   155→          history.back();
   156→          return;
   157→        } catch {
   158→          /* fallback زیر */
   159→        }
   160→      }
   161→      if (name === "drawer") setDrawerOpen(false);
   162→      else if (name === "nc") {
   163→        setNcOpen(false);
   164→        setNc(0);
   165→      } else if (name === "cc") {
   166→        setCcOpen(false);
   167→        setCc(0);
   168→      } else closeApp();
   169→    },
   170→    [closeApp, setNc, setCc],
   171→  );
   172→
   173→  /* وضعیت جاری برای هندلر popstate (بدون stale closure) */
   174→  const stateRef = useRef({
   175→    drawerOpen: false,
   176→    ncOpen: false,
   177→    ccOpen: false,
   178→    app: null as null | { id: AppId; rect: AppRect },
   179→    appPhase: "open" as "opening" | "open" | "closing",
   180→  });
   181→  useEffect(() => {
   182→    stateRef.current = { drawerOpen, ncOpen, ccOpen, app, appPhase };
   183→  }, [drawerOpen, ncOpen, ccOpen, app, appPhase]);
   184→
   185→  useEffect(() => {
   186→    try {
   187→      if (history.state && (history.state as { sd?: string }).sd) history.replaceState(null, "");
   188→    } catch {
   189→      /* بی‌اهمیت */
   190→    }
   191→    const onPop = () => {
   192→      lastPushedRef.current = null;
   193→      const s = stateRef.current;
   194→      if (s.drawerOpen) setDrawerOpen(false);
   195→      else if (s.ncOpen) {
   196→        setNcOpen(false);
   197→        setNc(0);
   198→      } else if (s.ccOpen) {
   199→        setCcOpen(false);
   200→        setCc(0);
   201→      } else if (s.app && s.appPhase !== "closing") closeApp();
   202→    };
   203→    window.addEventListener("popstate", onPop);
   204→    return () => window.removeEventListener("popstate", onPop);
   205→  }, [closeApp, setNc, setCc]);
   206→
   207→  /* صفحات و پارالاکس */
   208→  const pagesRef = useRef<HTMLDivElement | null>(null);
   209→  const wallRef = useRef<HTMLDivElement | null>(null);
   210→  const deviceRef = useRef<HTMLDivElement | null>(null);
   211→  const [page, setPage] = useState(0);
   212→
   213→  const onPagesScroll = useCallback(() => {
   214→    const el = pagesRef.current;
   215→    if (!el) return;
   216→    const maxAbs = el.scrollWidth - el.clientWidth;
   217→    const p = maxAbs > 0 ? Math.min(1, Math.abs(el.scrollLeft) / maxAbs) : 0;
   218→    if (wallRef.current) {
   219→      wallRef.current.style.transform = `translateX(${(0.5 - p) * 44}px) scale(1.08)`;
   220→    }
   221→    setPage(Math.round(p * (PAGES - 1)));
   222→  }, []);
   223→
   224→  const jumpToPage = (i: number) => {
   225→    const el = pagesRef.current;
   226→    if (!el) return;
   227→    const maxAbs = el.scrollWidth - el.clientWidth;
   228→    el.scrollTo({ left: i === 0 ? 0 : -maxAbs, behavior: "smooth" });
   229→  };
   230→
   231→  /* حرکات کشیدن از بالا/پایین */
   232→  const gesture = useRef<{
   233→    type: "nc" | "cc" | "close-app";
   234→    startY: number;
   235→    id: number;
   236→  } | null>(null);
   237→
   238→  const onRootPointerDown = (e: React.PointerEvent) => {
   239→    if (wiggle || drag || gesture.current || drawerOpen || ncOpen || ccOpen) return;
   240→    const rect = deviceRef.current?.getBoundingClientRect();
   241→    if (!rect) return;
   242→    const y = e.clientY - rect.top;
   243→    const inApp = !!app && appPhase !== "closing";
   244→    if (y < 46 || y > rect.height - 44) {
   245→      /* لمس دکمه‌ها در نوار وضعیت/نوار خانه → کلیک مستقیم، نه حرکت */
   246→      const t = e.target as HTMLElement | null;
   247→      if (t?.closest?.("button,[role=button],a,select,input,textarea")) return;
   248→    }
   249→    if (y < 46) {
   250→      gesture.current = { type: "nc", startY: e.clientY, id: e.pointerId };
   251→      try {
   252→        deviceRef.current?.setPointerCapture(e.pointerId);
   253→      } catch {
   254→        /* بی‌اهمیت */
   255→      }
   256→    } else if (y > rect.height - 44) {
   257→      gesture.current = { type: inApp ? "close-app" : "cc", startY: e.clientY, id: e.pointerId };
   258→      try {
   259→        deviceRef.current?.setPointerCapture(e.pointerId);
   260→      } catch {
   261→        /* بی‌اهمیت */
   262→      }
   263→    }
   264→  };
   265→
   266→  const onRootPointerMove = (e: React.PointerEvent) => {
   267→    const g = gesture.current;
   268→    if (!g || e.pointerId !== g.id) return;
   269→    const dy = e.clientY - g.startY;
   270→    if (g.type === "nc") setNc(Math.max(0, Math.min(1, dy / 150)));
   271→    else if (g.type === "cc") setCc(Math.max(0, Math.min(1, -dy / 170)));
   272→  };
   273→
   274→  const onRootPointerUp = (e: React.PointerEvent) => {
   275→    const g = gesture.current;
   276→    gesture.current = null;
   277→    if (!g || e.pointerId !== g.id) return;
   278→    if (g.type === "nc") {
   279→      const open = ncProgRef.current > 0.4;
   280→      if (open) {
   281→        setNcOpen(true);
   282→        setNc(1);
   283→        pushLayer("nc");
   284→      } else setNc(0);
   285→    } else if (g.type === "cc") {
   286→      const open = ccProgRef.current > 0.4;
   287→      if (open) {
   288→        setCcOpen(true);
   289→        setCc(1);
   290→        pushLayer("cc");
   291→      } else setCc(0);
   292→    } else {
   293→      // کشیدن به بالا روی نوار خانه هنگام باز بودن اپ → بستن اپ
   294→      if (e.clientY - g.startY < -36 && app && appPhase !== "closing") closeLayerUI("app");
   295→    }
   296→  };
   297→
   298→  const openNC = () => {
   299→    setNcOpen(true);
   300→    setNc(1);
   301→    pushLayer("nc");
   302→  };
   303→  const openCC = () => {
   304→    setCcOpen(true);
   305→    setCc(1);
   306→    pushLayer("cc");
   307→  };
   308→  const openDrawer = () => {
   309→    if (drawerOpen) return;
   310→    if (ncOpen) closeLayerUI("nc");
   311→    if (ccOpen) closeLayerUI("cc");
   312→    setDrawerOpen(true);
   313→    pushLayer("drawer");
   314→  };
   315→  const navFromDrawer = (kind: "app" | "nc" | "cc", id?: AppId) => {
   316→    setDrawerOpen(false);
   317→    try {
   318→      history.replaceState({ sd: kind }, "");
   319→      lastPushedRef.current = kind;
   320→    } catch {
   321→      /* بی‌اهمیت */
   322→    }
   323→    buzz();
   324→    if (kind === "app" && id) launchApp(id, true);
   325→    else if (kind === "nc") {
   326→      setNcOpen(true);
   327→      setNc(1);
   328→    } else {
   329→      setCcOpen(true);
   330→      setCc(1);
   331→    }
   332→  };
   333→
   334→  /* ترتیب ذخیره‌شده آیکون‌ها */
   335→  useEffect(() => {
   336→    let alive = true;
   337→    (async () => {
   338→      try {
   339→        const raw = localStorage.getItem(GRID_ORDER_KEY);
   340→        if (raw) {
   341→          const arr = JSON.parse(raw) as AppId[];
   342→          if (Array.isArray(arr)) {
   343→            /* ادغام ترتیب ذخیره‌شده با اپ‌های جدید (اپ جدید به انتها اضافه می‌شود) */
   344→            const known = arr.filter((a) => a in APPS);
   345→            const merged = [...known, ...DEFAULT_GRID_ORDER.filter((d) => !known.includes(d))];
   346→            if (merged.length && alive) setGridOrder(merged);
   347→          }
   348→        }
   349→      } catch {
   350→        /* ترتیب پیش‌فرض */
   351→      }
   352→    })();
   353→    return () => {
   354→      alive = false;
   355→    };
   356→  }, []);
   357→
   358→  const persistOrder = (order: AppId[]) => {
   359→    try {
   360→      localStorage.setItem(GRID_ORDER_KEY, JSON.stringify(order));
   361→    } catch {
   362→      /* حالت ناشناس */
   363→    }
   364→  };
   365→
   366→  /* لانگ‌پرس → لرزش | در حالت لرزش → کشیدن */
   367→  const pressRef = useRef<{
   368→    id: AppId;
   369→    startX: number;
   370→    startY: number;
   371→    timer: ReturnType<typeof setTimeout> | null;
   372→    longFired: boolean;
   373→  } | null>(null);
   374→  const iconRefs = useRef<Partial<Record<AppId, HTMLButtonElement | null>>>({});
   375→
   376→  const cancelLongPress = () => {
   377→    if (pressRef.current?.timer) clearTimeout(pressRef.current.timer);
   378→  };
   379→
   380→  const startLongPress = (id: AppId, e: React.PointerEvent) => {
   381→    const el = e.currentTarget as HTMLElement;
   382→    try {
   383→      el.setPointerCapture(e.pointerId);
   384→    } catch {
   385→      /* بی‌اهمیت */
   386→    }
   387→    pressRef.current = {
   388→      id,
   389→      startX: e.clientX,
   390→      startY: e.clientY,
   391→      timer: setTimeout(() => {
   392→        setWiggle(true);
   393→        buzz();
   394→        if (pressRef.current) pressRef.current.longFired = true;
   395→      }, LONG_PRESS_MS),
   396→      longFired: false,
   397→    };
   398→  };
   399→
   400→  const startDrag = (id: AppId, e: React.PointerEvent) => {
   401→    const el = e.currentTarget as HTMLElement;
   402→    try {
   403→      el.setPointerCapture(e.pointerId);
   404→    } catch {
   405→      /* بی‌اهمیت */
   406→    }
   407→    setDrag({ id, x: e.clientX, y: e.clientY });
   408→  };
   409→
   410→  const onIconPointerMove = (id: AppId, e: React.PointerEvent) => {
   411→    if (wiggle) {
   412→      if (!drag || drag.id !== id) return;
   413→      setDrag({ id, x: e.clientX, y: e.clientY });
   414→      const target = gridOrder.findIndex((a) => {
   415→        const el = iconRefs.current[a];
   416→        if (!el) return false;
   417→        const r = el.getBoundingClientRect();
   418→        return e.clientX >= r.left && e.clientX <= r.right;
   419→      });
   420→      if (target >= 0 && target !== gridOrder.indexOf(id)) {
   421→        setGridOrder((prev) => {
   422→          const from = prev.indexOf(id);
   423→          const next = [...prev];
   424→          next.splice(from, 1);
   425→          next.splice(target, 0, id);
   426→          return next;
   427→        });
   428→      }
   429→      return;
   430→    }
   431→    const p = pressRef.current;
   432→    if (p && p.id === id) {
   433→      const dist = Math.hypot(e.clientX - p.startX, e.clientY - p.startY);
   434→      if (dist > 10) cancelLongPress();
   435→    }
   436→  };
   437→
   438→  const onIconPointerUp = () => {
   439→    cancelLongPress();
   440→    if (drag) {
   441→      persistOrder(gridOrder);
   442→      setDrag(null);
   443→      buzz();
   444→    }
   445→  };
   446→
   447→  const launchApp = (id: AppId, noPush = false) => {
   448→    const el = iconRefs.current[id];
   449→    const dRect = deviceRef.current?.getBoundingClientRect();
   450→    if (!dRect) return;
   451→    const r = el?.getBoundingClientRect() ?? dRect;
   452→    buzz();
   453→    setApp({
   454→      id,
   455→      rect: {
   456→        x: r.left - dRect.left,
   457→        y: r.top - dRect.top,
   458→        w: r.width,
   459→        h: r.height,
   460→        dw: dRect.width,
   461→        dh: dRect.height,
   462→      },
   463→    });
   464→    setAppPhase("opening");
   465→    requestAnimationFrame(() => requestAnimationFrame(() => setAppPhase("open")));
   466→    if (!noPush) pushLayer("app");
   467→  };
   468→
   469→  const onIconClick = (id: AppId) => {
   470→    if (pressRef.current?.longFired) {
   471→      if (pressRef.current) pressRef.current.longFired = false;
   472→      return;
   473→    }
   474→    if (wiggle) return;
   475→    launchApp(id);
   476→  };
   477→
   478→  /* بستن لایه بالایی با Esc */
   479→  useEffect(() => {
   480→    const onKey = (e: KeyboardEvent) => {
   481→      if (e.key !== "Escape") return;
   482→      if (drawerOpen) closeLayerUI("drawer");
   483→      else if (ncOpen) closeLayerUI("nc");
   484→      else if (ccOpen) closeLayerUI("cc");
   485→      else if (app) closeLayerUI("app");
   486→    };
   487→    window.addEventListener("keydown", onKey);
   488→    return () => window.removeEventListener("keydown", onKey);
   489→  }, [drawerOpen, ncOpen, ccOpen, app, closeLayerUI]);
   490→
   491→  /* راهنمای اولیه */
   492→  const [hint, setHint] = useState(false);
   493→  useEffect(() => {
   494→    let alive = true;
   495→    (async () => {
   496→      await Promise.resolve();
   497→      if (!alive) return;
   498→      try {
   499→        if (!localStorage.getItem(SHELL_HINT_KEY)) {
   500→          if (alive) setHint(true);
   501→          localStorage.setItem(SHELL_HINT_KEY, "1");
   502→        }
   503→      } catch {
   504→        /* حالت ناشناس */
   505→      }
   506→    })();
   507→    return () => {
   508→      alive = false;
   509→    };
   510→  }, []);
   511→
   512→  /* ---------- محتوای اپ‌ها ---------- */
   513→
   514→  const appContent = (() => {
   515→    if (!app) return null;
   516→    if (app.id === "news") return <NewsApp />;
   517→    if (app.id === "about") return <AboutApp />;
   518→    if (app.id === "chart") return <ChartApp />;
   519→    if (app.id === "structure")
   520→      return (
   521→        <div className="p-4">
   522→          <MarketStructurePanel
   523→            data={ms}
   524→            refreshing={msRefreshing}
   525→            onExternalRefresh={() => refreshMarket(true)}
   526→          />
   527→        </div>
   528→      );
   529→    return (
   530→      <div className="p-4">
   531→        {ms ? (
   532→          <TradeSuggestion data={ms} />
   533→        ) : (
   534→          <div className="flex flex-col gap-2">
   535→            <Skeleton className="h-10 w-full" />
   536→            <Skeleton className="h-40 w-full" />
   537→          </div>
   538→        )}
   539→      </div>
   540→    );
   541→  })();
   542→
   543→  const sheetTransform =
   544→    app && appPhase !== "open"
   545→      ? `translate(${app.rect.x}px, ${app.rect.y}px) scale(${app.rect.w / app.rect.dw}, ${app.rect.h / app.rect.dh})`
   546→      : "none";
   547→
   548→  return (
   549→    <div
   550→      ref={deviceRef}
   551→      onPointerDown={onRootPointerDown}
   552→      onPointerMove={onRootPointerMove}
   553→      onPointerUp={onRootPointerUp}
   554→      onPointerCancel={() => {
   555→        gesture.current = null;
   556→        setNc(ncOpen ? 1 : 0);
   557→        setCc(ccOpen ? 1 : 0);
   558→      }}
