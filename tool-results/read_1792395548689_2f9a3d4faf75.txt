    12→    12→/** بازده لگاریتمی ساده بین مقادیر متوالی */
    13→    13→export function pctReturns(closes: number[]): number[] {
    14→    14→  const out: number[] = [];
    15→    15→  for (let i = 1; i < closes.length; i++) {
    16→    16→    const a = closes[i - 1];
    17→    17→    if (a !== 0 && isFinite(a) && isFinite(closes[i])) out.push((closes[i] - a) / a);
    18→    18→  }
    19→    19→  return out;
    20→    20→}

    22→    22→/** ضریب همبستگی پیرسون بین دو سری هم‌طول */
    23→    23→export function pearson(a: number[], b: number[]): number | null {
    24→    24→  const n = Math.min(a.length, b.length);
    25→    25→  if (n < 20) return null;
    26→    26→  let sa = 0;
    27→    27→  let sb = 0;
    28→    28→  for (let i = 0; i < n; i++) {
    29→    29→    sa += a[i];
    30→    30→    sb += b[i];
    31→    31→  }
    32→    32→  const ma = sa / n;
    33→    33→  const mb = sb / n;
    34→    34→  let num = 0;
    35→    35→  let da = 0;
    36→    36→  let dbb = 0;
    37→    37→  for (let i = 0; i < n; i++) {
    38→    38→    const x = a[i] - ma;
    39→    39→    const y = b[i] - mb;
    40→    40→    num += x * y;
    41→    41→    da += x * x;
    42→    42→    dbb += y * y;
    43→    43→  }
    44→    44→  if (da === 0 || dbb === 0) return null;
    45→    45→  return num / Math.sqrt(da * dbb);
    46→    46→}

    48→    48→/** ATR درصدی (میانگین دامنه واقعی نسبت به قیمت) */
    49→    49→export function atrPct(
    50→    50→  highs: number[],
    51→    51→  lows: number[],
    52→    52→  closes: number[],
    53→    53→  period = 14
    54→    54→): number | null {
    55→    55→  const n = closes.length;
    56→    56→  if (n < period + 1) return null;
    57→    57→  const trs: number[] = [];
    58→    58→  for (let i = 1; i < n; i++) {
    59→    59→    trs.push(
    60→    60→      Math.max(
    61→    61→        highs[i] - lows[i],
    62→    62→        Math.abs(highs[i] - closes[i - 1]),
    63→    63→        Math.abs(lows[i] - closes[i - 1])
    64→    64→      )
    65→    65→    );
    66→    66→  }
    67→    67→  let a = 0;
    68→    68→  for (let i = trs.length - period; i < trs.length; i++) a += trs[i];
    69→    69→  a /= period;
    70→    70→  return (a / closes[n - 1]) * 100;
    71→    71→}

    73→    73→/** نسبت کارایی (Efficiency Ratio) — معیار رونددار بودن بازار */
    74→    74→export function efficiencyRatio(closes: number[], window = 20): number | null {
    75→    75→  if (closes.length < window + 1) return null;
    76→    76→  const seg = closes.slice(-(window + 1));
    77→    77→  const net = Math.abs(seg[seg.length - 1] - seg[0]);
    78→    78→  let path = 0;
    79→    79→  for (let i = 1; i < seg.length; i++) path += Math.abs(seg[i] - seg[i - 1]);
    80→    80→  if (path === 0) return null;
    81→    81→  return net / path;
    82→    82→}

   322→   322→export interface PlanOutput {
   323→   323→  ok: boolean;
   324→   324→  error?: string;
   325→   325→  sl: number | null;
   326→   326→  tp: number | null;
   327→   327→  slDist: number | null;
   328→   328→  tpDist: number | null;
   329→   329→  riskAmount: number | null;
   330→   330→  lots: number | null;
   331→   331→  lossAtSL: number | null;
   332→   332→  profitAtTP: number | null;
   333→   333→  copyText: string;
   334→   334→  warnings: string[];
   335→   335→}
   336→   336→
   337→   337→export function computeTradePlan(input: PlanInput): PlanOutput {
   338→   338→  const warnings: string[] = [];
   339→   339→  const base: PlanOutput = {
   340→   340→    ok: false,
   341→   341→    sl: null,
   342→   342→    tp: null,
   343→   343→    slDist: null,
   344→   344→    tpDist: null,
   345→   345→    riskAmount: null,
   346→   346→    lots: null,
   347→   347→    lossAtSL: null,
   348→   348→    profitAtTP: null,
   349→   349→    copyText: "",
   350→   350→    warnings,
   351→   351→  };
   352→   352→  const { side, entry, balance, riskPct, rr, slMode, swingLevel, atr, atrMult, contractSize } = input;
   353→   353→  if (!(entry > 0)) return { ...base, error: "قیمت ورود معتبر نیست." };
   354→   354→  if (!(balance > 0)) return { ...base, error: "موجودی حساب معتبر نیست." };
   355→   355→  if (!(riskPct > 0 && riskPct <= 10)) return { ...base, error: "درصد ریسک باید بین صفر و ۱۰ باشد." };
   356→   356→  if (contractSize <= 0)
   357→   357→    return { ...base, error: "اندازه قرارداد نامعتبر است — مشخصات کارگزاری را تأیید کنید." };
   358→   358→
   359→   359→  let slDist: number | null = null;
   360→   360→  if (slMode === "swing") {
   361→   361→    if (swingLevel === null || !(swingLevel > 0)) {
   362→   362→      warnings.push("سطح سوئینگ در دسترس نیست — حد ضرر بر اساس ATR پیشنهاد شد.");
   363→   363→    } else {
   364→   364→      slDist = Math.abs(entry - swingLevel);
   365→   365→      if (slDist <= 0) warnings.push("سوئینگ با قیمت ورود برابر است — از حالت ATR استفاده کنید.");
   366→   366→    }
   367→   367→  }
   368→   368→  if (slDist === null) {
   369→   369→    if (atr === null || !(atr > 0)) return { ...base, error: "ATR در دسترس نیست — حد ضرر قابل محاسبه نیست." };
   370→   370→    slDist = atr * atrMult;
   371→   371→  }
   372→   372→  const riskAmount = (balance * riskPct) / 100;
   373→   373→  const lotsRaw = riskAmount / (slDist * contractSize);
   374→   374→  const lots = Math.max(0.01, Math.floor(lotsRaw * 100) / 100);
   375→   375→  const lossAtSL = lots * slDist * contractSize;
   376→   376→  const tpDist = slDist * rr;
   377→   377→  const sl = side === "buy" ? entry - slDist : entry + slDist;
   378→   378→  const tp = side === "buy" ? entry + tpDist : entry - tpDist;
   379→   379→  const profitAtTP = lossAtSL * rr;
   380→   380→  const copyText = `${side === "buy" ? "Buy" : "Sell"} ${input.symbolLabel} @ ${entry} | SL: ${

   467→   467→export function refreshNudge(timestamps: number[], now = Date.now()): { count: number; warn: boolean; message: string } {

   480→   480→export function revengeLock(lastLossAtISO: string | null, now = Date.now()): { locked: boolean; minutesLeft: number } {
   481→   481→  if (!lastLossAtISO) return { locked: false, minutesLeft: 0 };
   482→   482→  const t = new Date(lastLossAtISO).getTime();
   483→   483→  if (!isFinite(t)) return { locked: false, minutesLeft: 0 };
   484→   484→  const left = Math.ceil((t + NUDGE_RULES.revengeLockMs - now) / 60000);
   485→   485→  return left > 0 ? { locked: true, minutesLeft: left } : { locked: false, minutesLeft: 0 };
   486→   486→}

   560→   560→export function diffSinceLastVisit(prev: VisitSnapshot, cur: VisitSnapshot): VisitChange[] {
   561→   561→  const out: VisitChange[] = [];
   562→   562→  for (const key of Object.keys(cur.trends)) {
   563→   563→    const before = prev.trends?.[key];
   564→   564→    const after = cur.trends[key];
   565→   565→    if (before && after && before !== after) {
   566→   566→      const [sym, tf] = key.split(":");
   567→   567→      const symFa = sym === "xau" ? "طلا" : "داوجونز";
   568→   568→      const tfFa: Record<string, string> = {
   569→   569→        "15m": "۱۵ دقیقه",
   570→   570→        "1h": "۱ ساعته",
   571→   571→        "4h": "۴ ساعته",
   572→   572→        "1d": "روزانه",
   573→   573→      };
   574→   574→      const dirFa: Record<string, string> = { bullish: "صعودی", bearish: "نزولی", neutral: "خنثی" };
   575→   575→      out.push({
   576→   576→        kind: "trend",
   577→   577→        text: `جهت ساختار ${symFa} در تایم‌فریم ${tfFa[tf] ?? tf} از «${dirFa[before] ?? before}» به «${dirFa[after] ?? after}» تغییر کرد.`,
   578→   578→      });
   579→   579→    }
   580→   580→  }
   581→   581→  if (prev.score !== null && cur.score !== null && Math.abs(cur.score - prev.score) >= 10) {
   582→   582→    out.push({
   583→   583→      kind: "score",
   584→   584→      text: `امتیاز اطمینان از ${prev.score} به ${cur.score} تغییر کرد.`,
   585→   585→    });
   586→   586→  }

   588→   588→    out.push({
   589→   589→      kind: "divergence",
   590→   590→      text: `وضعیت همبستگی با DXY از «${prev.divergence}» به «${cur.divergence}» تغییر کرد.`,
   591→   591→    });
   592→   592→  }
   593→   593→  if (prev.highEventId && cur.highEventId && prev.highEventId !== cur.highEventId) {
   594→   594→    out.push({ kind: "event", text: "خبر مهم جدیدی به تقویم اضافه یا جایگزین خبر قبلی شد." });
   595→   595→  }
   596→   596→  if (prev.price !== null && cur.price !== null && prev.price > 0) {
   597→   597→    const pct = ((cur.price - prev.price) / prev.price) * 100;
   598→   598→    if (Math.abs(pct) >= 0.5) {
   599→   599→      out.push({
   600→   600→        kind: "price",
   601→   601→        text: `قیمت نماد نسبت به آخرین بازدید ${pct > 0 ? "+" : ""}${Math.round(pct * 100) / 100}٪ تغییر کرده است.`,
   602→   602→      });
   603→   603→    }
   604→   604→  }
   605→   605→  return out;
   606→   606→}
