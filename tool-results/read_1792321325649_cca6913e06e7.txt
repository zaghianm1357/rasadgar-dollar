    81→  try {
    82→    navigator.vibrate?.(12);
    83→  } catch {
    84→    /* پشتیبانی نمی‌شود */
    85→  }

    98→
    99→function Shell() {
   100→  const { ms, msRefreshing, refreshMarket, now } = useShell();
   101→
   102→  /* حالت لرزش و جابه‌جایی آیکون‌ها */
   103→  const [wiggle, setWiggle] = useState(false);
   104→  const [gridOrder, setGridOrder] = useState<AppId[]>(DEFAULT_GRID_ORDER);
   105→  const [drag, setDrag] = useState<{ id: AppId; x: number; y: number } | null>(null);
   106→
   107→  /* مراکز اعلان و کنترل */
   108→  const [ncOpen, setNcOpen] = useState(false);
   109→  const [ccOpen, setCcOpen] = useState(false);
   110→  const [ncProgress, setNcProgress] = useState(0);
   111→  const [ccProgress, setCcProgress] = useState(0);
   112→  const ncProgRef = useRef(0);
   113→  const ccProgRef = useRef(0);
   114→
   115→  /* منوی جانبی */
   116→  const [drawerOpen, setDrawerOpen] = useState(false);
   117→
   118→  const setNc = useCallback((v: number) => {
   119→    ncProgRef.current = v;
   120→    setNcProgress(v);
   121→  }, []);
   122→  const setCc = useCallback((v: number) => {
   123→    ccProgRef.current = v;
   124→    setCcProgress(v);
   125→  }, []);
   126→
   127→  /* اجرای اپ با انیمیشن بزرگ‌نمایی از موقعیت آیکون (FLIP) */
   128→  const [app, setApp] = useState<{ id: AppId; rect: AppRect } | null>(null);
   129→  const [appPhase, setAppPhase] = useState<"opening" | "open" | "closing">("open");
   130→
   131→  /* ---------- لایه‌های تاریخچه — دکمه برگشت خود گوشی ---------- */
   132→  const lastPushedRef = useRef<string | null>(null);
   133→  const pushLayer = useCallback((name: string) => {
   134→    lastPushedRef.current = name;
   135→    try {
   136→      history.pushState({ sd: name }, "");
   137→    } catch {
   138→      lastPushedRef.current = null;
   139→    }
   140→  }, []);
   141→
   142→  const closeApp = useCallback(() => {
   143→    setAppPhase("closing");
   144→    setTimeout(() => setApp(null), 420);
   145→  }, []);
   146→
   147→  const closeLayerUI = useCallback(
   148→    (name: "app" | "nc" | "cc" | "drawer") => {
   149→      if (lastPushedRef.current === name) {
   150→        lastPushedRef.current = null;
   151→        try {
   152→          history.back();
   153→          return;
   154→        } catch {
   155→          /* fallback زیر */
   156→        }
   157→      }
   158→      if (name === "drawer") setDrawerOpen(false);
   159→      else if (name === "nc") {
   160→        setNcOpen(false);
   161→        setNc(0);
   162→      } else if (name === "cc") {
   163→        setCcOpen(false);
   164→        setCc(0);
   165→      } else closeApp();
   166→    },
   167→    [closeApp, setNc, setCc],
   168→  );
   169→
   170→  /* وضعیت جاری برای هندلر popstate (بدون stale closure) */
   171→  const stateRef = useRef({
   172→    drawerOpen: false,
   173→    ncOpen: false,
   174→    ccOpen: false,
   175→    app: null as null | { id: AppId; rect: AppRect },
   176→    appPhase: "open" as "opening" | "open" | "closing",
   177→  });
   178→  useEffect(() => {
   179→    stateRef.current = { drawerOpen, ncOpen, ccOpen, app, appPhase };
   180→  }, [drawerOpen, ncOpen, ccOpen, app, appPhase]);
   181→
   182→  useEffect(() => {
   183→    try {
   184→      if (history.state && (history.state as { sd?: string }).sd) history.replaceState(null, "");
   185→    } catch {
   186→      /* بی‌اهمیت */
   187→    }
   188→    const onPop = () => {
   189→      lastPushedRef.current = null;
   190→      const s = stateRef.current;
   191→      if (s.drawerOpen) setDrawerOpen(false);
   192→      else if (s.ncOpen) {
   193→        setNcOpen(false);
   194→        setNc(0);
   195→      } else if (s.ccOpen) {
   196→        setCcOpen(false);
   197→        setCc(0);
   198→      } else if (s.app && s.appPhase !== "closing") closeApp();
   199→    };
   200→    window.addEventListener("popstate", onPop);
   201→    return () => window.removeEventListener("popstate", onPop);
   202→  }, [closeApp, setNc, setCc]);
   203→
   204→  /* صفحات و پارالاکس */
   205→  const pagesRef = useRef<HTMLDivElement | null>(null);
   206→  const wallRef = useRef<HTMLDivElement | null>(null);
   207→  const deviceRef = useRef<HTMLDivElement | null>(null);
   208→  const [page, setPage] = useState(0);
   209→
   210→  const onPagesScroll = useCallback(() => {
   211→    const el = pagesRef.current;
   212→    if (!el) return;
   213→    const maxAbs = el.scrollWidth - el.clientWidth;
   214→    const p = maxAbs > 0 ? Math.min(1, Math.abs(el.scrollLeft) / maxAbs) : 0;
   215→    if (wallRef.current) {
   216→      wallRef.current.style.transform = `translateX(${(0.5 - p) * 44}px) scale(1.08)`;
   217→    }
   218→    setPage(Math.round(p * (PAGES - 1)));
   219→  }, []);
   220→
   221→  const jumpToPage = (i: number) => {
   222→    const el = pagesRef.current;
   223→    if (!el) return;
   224→    const maxAbs = el.scrollWidth - el.clientWidth;
   225→    el.scrollTo({ left: i === 0 ? 0 : -maxAbs, behavior: "smooth" });
   226→  };
   227→
   228→  /* حرکات کشیدن از بالا/پایین */
   229→  const gesture = useRef<{
   230→    type: "nc" | "cc" | "close-app";
   231→    startY: number;
   232→    id: number;
   233→  } | null>(null);
   234→
   235→  const onRootPointerDown = (e: React.PointerEvent) => {
   236→    if (wiggle || drag || gesture.current || drawerOpen || ncOpen || ccOpen) return;
   237→    const rect = deviceRef.current?.getBoundingClientRect();
   238→    if (!rect) return;
   239→    const y = e.clientY - rect.top;
   240→    const inApp = !!app && appPhase !== "closing";
   241→    if (y < 46 || y > rect.height - 44) {
   242→      /* لمس دکمه‌ها در نوار وضعیت/نوار خانه → کلیک مستقیم، نه حرکت */
   243→      const t = e.target as HTMLElement | null;
   244→      if (t?.closest?.("button,[role=button],a,select,input,textarea")) return;
   245→    }
   246→    if (y < 46) {
   247→      gesture.current = { type: "nc", startY: e.clientY, id: e.pointerId };
   248→      try {
   249→        deviceRef.current?.setPointerCapture(e.pointerId);
   250→      } catch {
   251→        /* بی‌اهمیت */
   252→      }
   253→    } else if (y > rect.height - 44) {
   254→      gesture.current = { type: inApp ? "close-app" : "cc", startY: e.clientY, id: e.pointerId };
   255→      try {
   256→        deviceRef.current?.setPointerCapture(e.pointerId);
   257→      } catch {
   258→        /* بی‌اهمیت */
   259→      }
   260→    }
   261→  };
   262→
   263→  const onRootPointerMove = (e: React.PointerEvent) => {
   264→    const g = gesture.current;
   265→    if (!g || e.pointerId !== g.id) return;
   266→    const dy = e.clientY - g.startY;
   267→    if (g.type === "nc") setNc(Math.max(0, Math.min(1, dy / 150)));
   268→    else if (g.type === "cc") setCc(Math.max(0, Math.min(1, -dy / 170)));
   269→  };
   270→
   271→  const onRootPointerUp = (e: React.PointerEvent) => {
   272→    const g = gesture.current;
   273→    gesture.current = null;
   274→    if (!g || e.pointerId !== g.id) return;
   275→    if (g.type === "nc") {
   276→      const open = ncProgRef.current > 0.4;
   277→      if (open) {
   278→        setNcOpen(true);
   279→        setNc(1);
   280→        pushLayer("nc");
   281→      } else setNc(0);
   282→    } else if (g.type === "cc") {
   283→      const open = ccProgRef.current > 0.4;
   284→      if (open) {
   285→        setCcOpen(true);
   286→        setCc(1);
   287→        pushLayer("cc");
   288→      } else setCc(0);
   289→    } else {
   290→      // کشیدن به بالا روی نوار خانه هنگام باز بودن اپ → بستن اپ
   291→      if (e.clientY - g.startY < -36 && app && appPhase !== "closing") closeLayerUI("app");
   292→    }
   293→  };
   294→
   295→  const openNC = () => {
   296→    setNcOpen(true);
   297→    setNc(1);
   298→    pushLayer("nc");
   299→  };
   300→  const openCC = () => {
   301→    setCcOpen(true);
   302→    setCc(1);
   303→    pushLayer("cc");
   304→  };
   305→  const openDrawer = () => {
   306→    if (drawerOpen) return;
   307→    if (ncOpen) closeLayerUI("nc");
   308→    if (ccOpen) closeLayerUI("cc");
   309→    setDrawerOpen(true);
   310→    pushLayer("drawer");
   311→  };
   312→  const navFromDrawer = (kind: "app" | "nc" | "cc", id?: AppId) => {
   313→    setDrawerOpen(false);
   314→    try {
   315→      history.replaceState({ sd: kind }, "");
   316→      lastPushedRef.current = kind;
   317→    } catch {
   318→      /* بی‌اهمیت */
   319→    }
   320→    buzz();
   321→    if (kind === "app" && id) launchApp(id, true);
   322→    else if (kind === "nc") {
   323→      setNcOpen(true);
   324→      setNc(1);
   325→    } else {
   326→      setCcOpen(true);
   327→      setCc(1);
   328→    }
   329→  };
   330→
   331→  /* ترتیب ذخیره‌شده آیکون‌ها */
   332→  useEffect(() => {
   333→    let alive = true;
   334→    (async () => {
   335→      try {
   336→        const raw = localStorage.getItem(GRID_ORDER_KEY);
   337→        if (raw) {
   338→          const arr = JSON.parse(raw) as AppId[];
   339→          if (Array.isArray(arr) && arr.every((a) => a in APPS) && alive) {
   340→            setGridOrder(arr);
   341→          }
   342→        }
   343→      } catch {
   344→        /* ترتیب پیش‌فرض */
   345→      }
   346→    })();
   347→    return () => {
   348→      alive = false;
   349→    };
   350→  }, []);
   351→
   352→  const persistOrder = (order: AppId[]) => {
   353→    try {
   354→      localStorage.setItem(GRID_ORDER_KEY, JSON.stringify(order));
   355→    } catch {
   356→      /* حالت ناشناس */
   357→    }
   358→  };
   359→
   360→  /* لانگ‌پرس → لرزش | در حالت لرزش → کشیدن */
   361→  const pressRef = useRef<{
   362→    id: AppId;
   363→    startX: number;
   364→    startY: number;
   365→    timer: ReturnType<typeof setTimeout> | null;
   366→    longFired: boolean;
   367→  } | null>(null);
   368→  const iconRefs = useRef<Partial<Record<AppId, HTMLButtonElement | null>>>({});
   369→
   370→  const cancelLongPress = () => {
   371→    if (pressRef.current?.timer) clearTimeout(pressRef.current.timer);
   372→  };
   373→
   374→  const startLongPress = (id: AppId, e: React.PointerEvent) => {
   375→    const el = e.currentTarget as HTMLElement;
   376→    try {
   377→      el.setPointerCapture(e.pointerId);
   378→    } catch {
   379→      /* بی‌اهمیت */
   380→    }
   381→    pressRef.current = {
   382→      id,
   383→      startX: e.clientX,
   384→      startY: e.clientY,
   385→      timer: setTimeout(() => {
   386→        setWiggle(true);
   387→        buzz();
   388→        if (pressRef.current) pressRef.current.longFired = true;
   389→      }, LONG_PRESS_MS),
   390→      longFired: false,
   391→    };
   392→  };
   393→
   394→  const startDrag = (id: AppId, e: React.PointerEvent) => {
   395→    const el = e.currentTarget as HTMLElement;
   396→    try {
   397→      el.setPointerCapture(e.pointerId);
   398→    } catch {
   399→      /* بی‌اهمیت */
   400→    }
   401→    setDrag({ id, x: e.clientX, y: e.clientY });
   402→  };
   403→
   404→  const onIconPointerMove = (id: AppId, e: React.PointerEvent) => {
   405→    if (wiggle) {
   406→      if (!drag || drag.id !== id) return;
   407→      setDrag({ id, x: e.clientX, y: e.clientY });
   408→      const target = gridOrder.findIndex((a) => {
   409→        const el = iconRefs.current[a];
   410→        if (!el) return false;
   411→        const r = el.getBoundingClientRect();
   412→        return e.clientX >= r.left && e.clientX <= r.right;
   413→      });
   414→      if (target >= 0 && target !== gridOrder.indexOf(id)) {
   415→        setGridOrder((prev) => {
   416→          const from = prev.indexOf(id);
   417→          const next = [...prev];
   418→          next.splice(from, 1);
   419→          next.splice(target, 0, id);
   420→          return next;
   421→        });
   422→      }
   423→      return;
   424→    }
   425→    const p = pressRef.current;
   426→    if (p && p.id === id) {
   427→      const dist = Math.hypot(e.clientX - p.startX, e.clientY - p.startY);
   428→      if (dist > 10) cancelLongPress();
   429→    }
   430→  };
   431→
   432→  const onIconPointerUp = () => {
   433→    cancelLongPress();
   434→    if (drag) {
   435→      persistOrder(gridOrder);
   436→      setDrag(null);
   437→      buzz();
   438→    }
   439→  };
   440→
   441→  const launchApp = (id: AppId, noPush = false) => {
   442→    const el = iconRefs.current[id];
   443→    const dRect = deviceRef.current?.getBoundingClientRect();
   444→    if (!dRect) return;
   445→    const r = el?.getBoundingClientRect() ?? dRect;
   446→    buzz();
   447→    setApp({
   448→      id,
   449→      rect: {
   450→        x: r.left - dRect.left,
   451→        y: r.top - dRect.top,
   452→        w: r.width,
   453→        h: r.height,
   454→        dw: dRect.width,
   455→        dh: dRect.height,
   456→      },
   457→    });
   458→    setAppPhase("opening");
   459→    requestAnimationFrame(() => requestAnimationFrame(() => setAppPhase("open")));
   460→    if (!noPush) pushLayer("app");
   461→  };
   462→
   463→  const onIconClick = (id: AppId) => {
   464→    if (pressRef.current?.longFired) {
   465→      if (pressRef.current) pressRef.current.longFired = false;
   466→      return;
   467→    }
   468→    if (wiggle) return;
   469→    launchApp(id);
   470→  };
   471→
   472→  /* بستن لایه بالایی با Esc */
   473→  useEffect(() => {
   474→    const onKey = (e: KeyboardEvent) => {
   475→      if (e.key !== "Escape") return;
   476→      if (drawerOpen) closeLayerUI("drawer");
   477→      else if (ncOpen) closeLayerUI("nc");
   478→      else if (ccOpen) closeLayerUI("cc");
   479→      else if (app) closeLayerUI("app");
   480→    };
   481→    window.addEventListener("keydown", onKey);
   482→    return () => window.removeEventListener("keydown", onKey);
   483→  }, [drawerOpen, ncOpen, ccOpen, app, closeLayerUI]);
   484→
   485→  /* راهنمای اولیه */
   486→  const [hint, setHint] = useState(false);
   487→  useEffect(() => {
   488→    let alive = true;
   489→    (async () => {
   490→      await Promise.resolve();
   491→      if (!alive) return;
   492→      try {
   493→        if (!localStorage.getItem(SHELL_HINT_KEY)) {
   494→          if (alive) setHint(true);
   495→          localStorage.setItem(SHELL_HINT_KEY, "1");
   496→        }
   497→      } catch {
   498→        /* حالت ناشناس */
   499→      }
   500→    })();
   501→    return () => {
   502→      alive = false;
   503→    };
   504→  }, []);
   505→
   506→  /* ---------- محتوای اپ‌ها ---------- */
   507→
   508→  const appContent = (() => {
   509→    if (!app) return null;
   510→    if (app.id === "news") return <NewsApp />;
   511→    if (app.id === "about") return <AboutApp />;
   512→    if (app.id === "structure")
   513→      return (
   514→        <div className="p-4">
   515→          <MarketStructurePanel
   516→            data={ms}
   517→            refreshing={msRefreshing}
   518→            onExternalRefresh={() => refreshMarket(true)}
   519→          />
   520→        </div>
   521→      );
   522→    return (
   523→      <div className="p-4">
   524→        {ms ? (
   525→          <TradeSuggestion data={ms} />
   526→        ) : (
   527→          <div className="flex flex-col gap-2">
   528→            <Skeleton className="h-10 w-full" />
   529→            <Skeleton className="h-40 w-full" />
   530→          </div>
   531→        )}
   532→      </div>
   533→    );
   534→  })();
   535→
   536→  const sheetTransform =
   537→    app && appPhase !== "open"
   538→      ? `translate(${app.rect.x}px, ${app.rect.y}px) scale(${app.rect.w / app.rect.dw}, ${app.rect.h / app.rect.dh})`
   539→      : "none";
   540→
   541→  return (
   542→    <div
   543→      ref={deviceRef}
   544→      onPointerDown={onRootPointerDown}
   545→      onPointerMove={onRootPointerMove}
   546→      onPointerUp={onRootPointerUp}
   547→      onPointerCancel={() => {
   548→        gesture.current = null;
   549→        setNc(ncOpen ? 1 : 0);
   550→        setCc(ccOpen ? 1 : 0);
   551→      }}
   552→      className="relative h-dvh w-full overflow-hidden bg-background select-none sm:h-[840px] sm:max-h-[94dvh] sm:w-[400px] sm:rounded-[3rem] sm:border sm:border-white/10 sm:shadow-[0_40px_90px_-24px_rgba(0,0,0,0.85)]"
   553→    >
   554→      {/* کاغذدیواری با پارالاکس */}
