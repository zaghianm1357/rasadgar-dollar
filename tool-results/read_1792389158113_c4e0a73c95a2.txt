   107→  var h = Math.floor(totalMin / 60), m = totalMin % 60;

   120→function newsGroups() {
   121→  var tKey = todayKeyInTehran();
   122→  var map = {}, order = [];
   123→  var evs = news.events || [];
   124→  for (var i = 0; i < evs.length; i++) {
   125→    var e = evs[i];
   126→    var f = S_NEWS.filter;
   127→    var ok = f === 'all' || (f === 'High' ? e.impact === 'High' : f === 'Medium' ? e.impact === 'Medium' : (e.impact === 'Low' || e.impact === 'Holiday'));
   128→    if (!ok) continue;
   129→    var k = e.dateISO ? dateKeyInTehran(new Date(e.dateISO)) : tKey;
   130→    if (!map[k]) { map[k] = []; order.push(k); }
   131→    map[k].push(e);
   132→  }
   133→  order.sort();
   134→  if (S_NEWS.tab === 'today') order = order.filter(function (k) { return k === tKey; });
   135→  else if (S_NEWS.tab === 'tomorrow') order = order.filter(function (k) { return k === addDaysToKey(tKey, 1); });
   136→  else if (S_NEWS.tab === 'week') {
   137→    var a = addDaysToKey(tKey, -6), b = addDaysToKey(tKey, 6);
   138→    order = order.filter(function (k) { return k >= a && k <= b; });
   139→  }
   140→  order.forEach(function (k) {
   141→    map[k].sort(function (x, y) {
   142→      return (x.dateISO ? new Date(x.dateISO).getTime() : -1) - (y.dateISO ? new Date(y.dateISO).getTime() : -1);
   143→    });
   144→  });
   145→  return order.map(function (k) { return [k, map[k]]; });
   146→}

   542→function emaLast(vals, n) {
   543→  if (!vals || vals.length < n + 5) return null;
   544→  var k = 2 / (n + 1), sum = 0;
   545→  for (var i = 0; i < n; i++) sum += vals[i];
   546→  var e = sum / n;
   547→  for (var i = n; i < vals.length; i++) e = vals[i] * k + e * (1 - k);
   548→  return e;
   549→}
   550→function rsiLast(c, n) {
   551→  if (!c || c.length < n + 1) return null;
   552→  var g = 0, l = 0, d, i;
   553→  for (i = 1; i <= n; i++) { d = c[i] - c[i - 1]; if (d > 0) g += d; else l -= d; }
   554→  var ag = g / n, al = l / n;
   555→  for (i = n + 1; i < c.length; i++) {
   556→    d = c[i] - c[i - 1];
   557→    ag = (ag * (n - 1) + (d > 0 ? d : 0)) / n;
   558→    al = (al * (n - 1) + (d < 0 ? -d : 0)) / n;
   559→  }
   560→  if (al === 0) return ag === 0 ? 50 : 100;
   561→  return 100 - 100 / (1 + ag / al);
   562→}
   563→function atrPct(h, l, c, n) {
   564→  if (!c || c.length < n + 1) return null;
   565→  var trs = [], i;
   566→  for (i = 1; i < c.length; i++) trs.push(Math.max(h[i] - l[i], Math.abs(h[i] - c[i - 1]), Math.abs(l[i] - c[i - 1])));
   567→  var a = 0;
   568→  for (i = 0; i < n; i++) a += trs[i];
   569→  a /= n;
   570→  for (i = n; i < trs.length; i++) a = (a * (n - 1) + trs[i]) / n;
   571→  return a / c[c.length - 1] * 100;
   572→}
   573→function findSwings(h, l, lb) {

   584→  }

   673→function runAi(force) {
   674→  var p = document.getElementById('ai_panel');
   675→  if (!p || AI_BUSY) return;
   676→  var key = aiKey();
   677→  var cached = AI_CACHE[key];
   678→  if (!force && cached && (Date.now() - cached.at) < AI_TTL) {
   679→    AI_STATE = { key: key, text: cached.text, at: cached.at };
   680→    aiRenderText();

   693→    AI_BUSY = false;
