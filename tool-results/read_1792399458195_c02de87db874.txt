    62→function lsGet<T>(key: string, fallback: T): T {
    63→  try {
    64→    const raw = localStorage.getItem(key);
    65→    if (!raw) return fallback;
    66→    return JSON.parse(raw) as T;
    67→  } catch {
    68→    return fallback;
    69→  }
    70→}
    71→function lsSet(key: string, v: unknown) {
    72→  try {
    73→    localStorage.setItem(key, JSON.stringify(v));
    74→  } catch {
    75→    /* حالت ناشناس */
    76→  }

    79→function faNum(v: number | null | undefined, digits = 2): string {
    80→  if (v === null || v === undefined || !isFinite(v)) return "—";
    81→  return faDigits(Number(v.toFixed(digits)).toLocaleString("en-US", { maximumFractionDigits: digits })).replace(/,/g, "٬");
    82→}
    83→function faPct(v: number | null | undefined): string {
    84→  if (v === null || v === undefined || !isFinite(v)) return "—";
    85→  return (v > 0 ? "+" : "") + faNum(v) + "٪";
    86→}

   215→            const cur = {

   219→              divergence: j.divergence?.level ?? null,

   221→              price: j.symbol?.price ?? null,
   222→            };
   223→            if (prev) {
   224→              const changes: { text: string }[] = [];
   225→              for (const key of Object.keys(cur.trends)) {
   226→                const before = prev.trends?.[key];
   227→                const after = cur.trends[key];
   228→                if (before && after && before !== after) {
   229→                  const [sk, tf] = key.split(":");
   230→                  changes.push({
   231→                    text: `جهت ساختار ${sk === "xau" ? "طلا" : "داوجونز"} در تایم‌فریم ${TF_FA[tf] ?? tf} از «${TREND_FA[before]}» به «${TREND_FA[after]}» تغییر کرد.`,
   232→                  });
   233→                }
   234→              }
   235→              if (prev.score !== null && cur.score !== null && Math.abs(cur.score - prev.score) >= 10)
   236→                changes.push({ text: `امتیاز اطمینان از ${faDigits(prev.score)} به ${faDigits(cur.score)} تغییر کرد.` });
   237→              if (prev.divergence && cur.divergence && prev.divergence !== cur.divergence)
   238→                changes.push({ text: cur.divergence === "normal" ? "رابطه معمول با DXY دوباره برقرار شد." : "وضعیت همبستگی با DXY تغییر کرد — پنل واگرایی را ببینید." });
   239→              if (prev.highEventId && cur.highEventId && prev.highEventId !== cur.highEventId)
   240→                changes.push({ text: "خبر مهم جدیدی به تقویم اضافه شد." });
   241→              if (prev.price !== null && cur.price !== null && prev.price > 0) {
   242→                const pct = ((cur.price - prev.price) / prev.price) * 100;
   243→                if (Math.abs(pct) >= 0.5) changes.push({ text: `قیمت نسبت به آخرین بازدید ${faPct(pct)} تغییر کرده است.` });
   244→              }
   245→              setVisitChanges(changes);
   246→            }
   247→            lsSet(K_VISIT, cur);

   249→            /* حالت ناشناس */
   250→          }
   251→        } else {
   252→          setError(String(j?.error ?? "خطای نامشخص"));
   253→        }
