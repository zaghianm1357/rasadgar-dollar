   409→   409→
   410→   410→  const onIconPointerMove = (id: AppId, e: React.PointerEvent) => {

   412→   412→      if (!drag || drag.id !== id) return;

   419→   419→      });

   422→   422→          const from = prev.indexOf(id);
   423→   423→          const next = [...prev];
   424→   424→          next.splice(from, 1);
   425→   425→          next.splice(target, 0, id);
   426→   426→          return next;
   427→   427→        });
   428→   428→      }
   429→   429→      return;
   430→   430→    }
   431→   431→    const p = pressRef.current;
   432→   432→    if (p && p.id === id) {
   433→   433→      const dist = Math.hypot(e.clientX - p.startX, e.clientY - p.startY);
   434→   434→      if (dist > 10) cancelLongPress();
   435→   435→    }
   436→   436→  };
   437→   437→
   438→   438→  const onIconPointerUp = () => {

   440→   440→    if (drag) {

   442→   442→      setDrag(null);
   443→   443→      buzz();
   444→   444→    }
   445→   445→  };

   449→   449→    const dRect = deviceRef.current?.getBoundingClientRect();
   450→   450→    if (!dRect) return;
   451→   451→    const r = el?.getBoundingClientRect() ?? dRect;
   452→   452→    buzz();
   453→   453→    setApp({
   454→   454→      id,
   455→   455→      rect: {
   456→   456→        x: r.left - dRect.left,
   457→   457→        y: r.top - dRect.top,
   458→   458→        w: r.width,
   459→   459→        h: r.height,
   460→   460→        dw: dRect.width,
   461→   461→        dh: dRect.height,
