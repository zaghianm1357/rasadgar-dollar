   113→  var totalMin = Math.max(0, Math.floor(msLeft / 60000));
   114→  var h = Math.floor(totalMin / 60), m = totalMin % 60;
   115→  if (h === 0) return faDigits(m) + ' دقیقه دیگر';
   116→  if (m === 0) return faDigits(h) + ' ساعت دیگر';
   117→  return faDigits(h) + ' ساعت و ' + faDigits(m) + ' دقیقه دیگر';

   127→function newsGroups() {
   128→  var tKey = todayKeyInTehran();
   129→  var map = {}, order = [];
   130→  var evs = news.events || [];
   131→  for (var i = 0; i < evs.length; i++) {
   132→    var e = evs[i];
   133→    var f = S_NEWS.filter;
   134→    var ok = f === 'all' || (f === 'High' ? e.impact === 'High' : f === 'Medium' ? e.impact === 'Medium' : (e.impact === 'Low' || e.impact === 'Holiday'));
   135→    if (!ok) continue;
   136→    var k = e.dateISO ? dateKeyInTehran(new Date(e.dateISO)) : tKey;
   137→    if (!map[k]) { map[k] = []; order.push(k); }
   138→    map[k].push(e);
   139→  }
   140→  order.sort();
   141→  if (S_NEWS.tab === 'today') order = order.filter(function (k) { return k === tKey; });
   142→  else if (S_NEWS.tab === 'tomorrow') order = order.filter(function (k) { return k === addDaysToKey(tKey, 1); });
   143→  else if (S_NEWS.tab === 'week') {
   144→    var a = addDaysToKey(tKey, -6), b = addDaysToKey(tKey, 6);
   145→    order = order.filter(function (k) { return k >= a && k <= b; });
   146→  }
   147→  order.forEach(function (k) {
   148→    map[k].sort(function (x, y) {
   149→      return (x.dateISO ? new Date(x.dateISO).getTime() : -1) - (y.dateISO ? new Date(y.dateISO).getTime() : -1);
   150→    });
   151→  });
   152→  return order.map(function (k) { return [k, map[k]]; });
   153→}

   549→function emaLast(vals, n) {
   550→  if (!vals || vals.length < n + 5) return null;
   551→  var k = 2 / (n + 1), sum = 0;
   552→  for (var i = 0; i < n; i++) sum += vals[i];
   553→  var e = sum / n;
   554→  for (var i = n; i < vals.length; i++) e = vals[i] * k + e * (1 - k);
   555→  return e;
   556→}
   557→function rsiLast(c, n) {
   558→  if (!c || c.length < n + 1) return null;
   559→  var g = 0, l = 0, d, i;
   560→  for (i = 1; i <= n; i++) { d = c[i] - c[i - 1]; if (d > 0) g += d; else l -= d; }
   561→  var ag = g / n, al = l / n;
   562→  for (i = n + 1; i < c.length; i++) {
   563→    d = c[i] - c[i - 1];
   564→    ag = (ag * (n - 1) + (d > 0 ? d : 0)) / n;
   565→    al = (al * (n - 1) + (d < 0 ? -d : 0)) / n;
   566→  }
   567→  if (al === 0) return ag === 0 ? 50 : 100;
   568→  return 100 - 100 / (1 + ag / al);
   569→}
   570→function atrPct(h, l, c, n) {
   571→  if (!c || c.length < n + 1) return null;
   572→  var trs = [], i;
   573→  for (i = 1; i < c.length; i++) trs.push(Math.max(h[i] - l[i], Math.abs(h[i] - c[i - 1]), Math.abs(l[i] - c[i - 1])));
   574→  var a = 0;
   575→  for (i = 0; i < n; i++) a += trs[i];
   576→  a /= n;
   577→  for (i = n; i < trs.length; i++) a = (a * (n - 1) + trs[i]) / n;
   578→  return a / c[c.length - 1] * 100;
   579→}

   704→function runAi(force) {
   705→  var p = document.getElementById('ai_panel');
   706→  if (!p || AI_BUSY) return;
   707→  var key = aiKey();
   708→  var cached = AI_CACHE[key];
   709→  if (!force && cached && (Date.now() - cached.at) < AI_TTL) {
   710→    AI_STATE = { key: key, text: cached.text, at: cached.at };
   711→    aiRenderText();
   712→    return;
   713→  }
   714→  var cfg = CHART_YAHOO[S_CHART.sym];
   715→  var itv = CHART_ITV[S_CHART.tf];
   716→  if (!cfg || !itv) { aiRenderError('این نماد در موتور تحلیل پشتیبانی نمی‌شود.'); return; }
   717→  AI_BUSY = true;
   718→  setAiBusy(true);
