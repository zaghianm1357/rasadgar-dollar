   417→   417→          const next = [...prev];
   418→   418→          next.splice(from, 1);
   419→   419→          next.splice(target, 0, id);
   420→   420→          return next;
   421→   421→        });
   422→   422→      }
   423→   423→      return;
   424→   424→    }
   425→   425→    const p = pressRef.current;

   438→   438→    }
   439→   439→  };

   449→   449→      rect: {
   450→   450→        x: r.left - dRect.left,
   451→   451→        y: r.top - dRect.top,
   452→   452→        w: r.width,
   453→   453→        h: r.height,
   454→   454→        dw: dRect.width,
   455→   455→        dh: dRect.height,
   456→   456→      },
   457→   457→    });

   461→   461→  };
   462→   462→
   463→   463→  const onIconClick = (id: AppId) => {

   467→   467→    }
   468→   468→    if (wiggle) return;
   469→   469→    launchApp(id);
