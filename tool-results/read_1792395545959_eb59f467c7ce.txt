    12→/** بازده لگاریتمی ساده بین مقادیر متوالی */
    13→export function pctReturns(closes: number[]): number[] {
    14→  const out: number[] = [];
    15→  for (let i = 1; i < closes.length; i++) {
    16→    const a = closes[i - 1];
    17→    if (a !== 0 && isFinite(a) && isFinite(closes[i])) out.push((closes[i] - a) / a);
    18→  }
    19→  return out;
    20→}

    22→/** ضریب همبستگی پیرسون بین دو سری هم‌طول */
    23→export function pearson(a: number[], b: number[]): number | null {
    24→  const n = Math.min(a.length, b.length);
    25→  if (n < 20) return null;
    26→  let sa = 0;
    27→  let sb = 0;
    28→  for (let i = 0; i < n; i++) {
    29→    sa += a[i];
    30→    sb += b[i];
    31→  }
    32→  const ma = sa / n;
    33→  const mb = sb / n;
    34→  let num = 0;
    35→  let da = 0;
    36→  let dbb = 0;
    37→  for (let i = 0; i < n; i++) {
    38→    const x = a[i] - ma;
    39→    const y = b[i] - mb;
    40→    num += x * y;
    41→    da += x * x;
    42→    dbb += y * y;
    43→  }
    44→  if (da === 0 || dbb === 0) return null;
    45→  return num / Math.sqrt(da * dbb);
    46→}

    48→/** ATR درصدی (میانگین دامنه واقعی نسبت به قیمت) */
    49→export function atrPct(
    50→  highs: number[],
    51→  lows: number[],
    52→  closes: number[],
    53→  period = 14
    54→): number | null {
    55→  const n = closes.length;
    56→  if (n < period + 1) return null;
    57→  const trs: number[] = [];
    58→  for (let i = 1; i < n; i++) {
    59→    trs.push(
    60→      Math.max(
    61→        highs[i] - lows[i],
    62→        Math.abs(highs[i] - closes[i - 1]),
    63→        Math.abs(lows[i] - closes[i - 1])
    64→      )
    65→    );
    66→  }
    67→  let a = 0;
    68→  for (let i = trs.length - period; i < trs.length; i++) a += trs[i];
    69→  a /= period;
    70→  return (a / closes[n - 1]) * 100;
    71→}

    73→/** نسبت کارایی (Efficiency Ratio) — معیار رونددار بودن بازار */
    74→export function efficiencyRatio(closes: number[], window = 20): number | null {
    75→  if (closes.length < window + 1) return null;
    76→  const seg = closes.slice(-(window + 1));
    77→  const net = Math.abs(seg[seg.length - 1] - seg[0]);
    78→  let path = 0;
    79→  for (let i = 1; i < seg.length; i++) path += Math.abs(seg[i] - seg[i - 1]);
    80→  if (path === 0) return null;
    81→  return net / path;
    82→}

   181→/** نوسان‌پذیری ATR (وزن ۱۰) — نوار سالم مخصوص هر نماد بر حسب درصد */
   182→export function atrFactor(
   183→  atrPctValue: number | null,
   184→  healthyBand: [number, number]
   185→): FactorResult {
   186→  if (atrPctValue === null) {
   187→    return { score: 5, label: "داده ناکافی", detail: "ATR محاسبه نشد." };
   188→  }
   189→  const v = Math.round(atrPctValue * 100) / 100;
   190→  if (atrPctValue < healthyBand[0] * 0.5) {
   191→    return {
   192→      score: 5,
   193→      label: "بازار رنج",
   194→      detail: `ATR روزانه ${v}٪ است — نوسان بسیار کم و بازار راکد.`,
   195→    };
   196→  }
   197→  if (atrPctValue > healthyBand[1] * 1.5) {
   198→    return {
   199→      score: 0,
   200→      label: "نوسان افراطی",
   201→      detail: `ATR روزانه ${v}٪ است — نوسان به‌شدت بالاتر از حالت معمول.`,
   202→    };
   203→  }
   204→  if (atrPctValue < healthyBand[0]) {
   205→    return { score: 5, label: "نوسان کم", detail: `ATR روزانه ${v}٪ است — نوسان پایین‌تر از نوار سالم.` };
   206→  }
   207→  if (atrPctValue > healthyBand[1]) {
   208→    return {
   209→      score: 5,
   210→      label: "نوسان بالا",
   211→      detail: `ATR روزانه ${v}٪ است — نوسان بالاتر از نوار سالم ولی افراطی نیست.`,
   212→    };
   213→  }
   214→  return { score: 10, label: "نوسان سالم", detail: `ATR روزانه ${v}٪ است — داخل نوار سالم.` };
   215→}

   308→export interface PlanInput {
   309→  side: "buy" | "sell";
   310→  entry: number;
   311→  balance: number;
   312→  riskPct: number;
   313→  rr: number; // ۲ یا ۳
   314→  slMode: "swing" | "atr";
   315→  swingLevel: number | null; // آخرین سوئینگ مناسب (کف برای خرید / سقف برای فروش)
   316→  atr: number | null; // مقدار ATR (قیمت — نه درصد)
   317→  atrMult: number;
   318→  contractSize: number; // ارزش هر یک واحد حرکت قیمت به ازای هر ۱ لات استاندارد
   319→  symbolLabel: string;
   320→}
   321→
   322→export interface PlanOutput {
   323→  ok: boolean;
   324→  error?: string;
   325→  sl: number | null;
   326→  tp: number | null;
   327→  slDist: number | null;
   328→  tpDist: number | null;
   329→  riskAmount: number | null;
   330→  lots: number | null;
   331→  lossAtSL: number | null;
   332→  profitAtTP: number | null;
   333→  copyText: string;
   334→  warnings: string[];
   335→}
   336→
   337→export function computeTradePlan(input: PlanInput): PlanOutput {
   338→  const warnings: string[] = [];
   339→  const base: PlanOutput = {
   340→    ok: false,
   341→    sl: null,
   342→    tp: null,
   343→    slDist: null,
   344→    tpDist: null,
   345→    riskAmount: null,
   346→    lots: null,
   347→    lossAtSL: null,
   348→    profitAtTP: null,
   349→    copyText: "",
   350→    warnings,
   351→  };
   352→  const { side, entry, balance, riskPct, rr, slMode, swingLevel, atr, atrMult, contractSize } = input;
   353→  if (!(entry > 0)) return { ...base, error: "قیمت ورود معتبر نیست." };
   354→  if (!(balance > 0)) return { ...base, error: "موجودی حساب معتبر نیست." };
   355→  if (!(riskPct > 0 && riskPct <= 10)) return { ...base, error: "درصد ریسک باید بین صفر و ۱۰ باشد." };
   356→  if (contractSize <= 0)
   357→    return { ...base, error: "اندازه قرارداد نامعتبر است — مشخصات کارگزاری را تأیید کنید." };
   358→
   359→  let slDist: number | null = null;
   360→  if (slMode === "swing") {
   361→    if (swingLevel === null || !(swingLevel > 0)) {
   362→      warnings.push("سطح سوئینگ در دسترس نیست — حد ضرر بر اساس ATR پیشنهاد شد.");
   363→    } else {
   364→      slDist = Math.abs(entry - swingLevel);
   365→      if (slDist <= 0) warnings.push("سوئینگ با قیمت ورود برابر است — از حالت ATR استفاده کنید.");
   366→    }
   367→  }
   368→  if (slDist === null) {
   369→    if (atr === null || !(atr > 0)) return { ...base, error: "ATR در دسترس نیست — حد ضرر قابل محاسبه نیست." };
   370→    slDist = atr * atrMult;
   371→  }
   372→  const riskAmount = (balance * riskPct) / 100;
   373→  const lotsRaw = riskAmount / (slDist * contractSize);
   374→  const lots = Math.max(0.01, Math.floor(lotsRaw * 100) / 100);
   375→  const lossAtSL = lots * slDist * contractSize;
   376→  const tpDist = slDist * rr;
   377→  const sl = side === "buy" ? entry - slDist : entry + slDist;
   378→  const tp = side === "buy" ? entry + tpDist : entry - tpDist;
   379→  const profitAtTP = lossAtSL * rr;
   380→  const copyText = `${side === "buy" ? "Buy" : "Sell"} ${input.symbolLabel} @ ${entry} | SL: ${
   381→    Math.round(sl * 100) / 100
   382→  } | TP: ${Math.round(tp * 100) / 100} | Lot: ${lots} (Risk ${riskPct}%)`;
   383→  warnings.push("اندازه قرارداد و ارزش پیپ به کارگزاری بستگی دارد — پیش از استفاده تأیید کنید.");
   384→  warnings.push("این محاسبه تضمین نتیجه یا توصیه معامله نیست.");
   385→  return {
   386→    ok: true,
   387→    sl: Math.round(sl * 100) / 100,
   388→    tp: Math.round(tp * 100) / 100,
   389→    slDist: Math.round(slDist * 100) / 100,
   390→    tpDist: Math.round(tpDist * 100) / 100,
   391→    riskAmount: Math.round(riskAmount * 100) / 100,
   392→    lots,
   393→    lossAtSL: Math.round(lossAtSL * 100) / 100,
   394→    profitAtTP: Math.round(profitAtTP * 100) / 100,
   395→    copyText,
   396→    warnings,
   397→  };
   398→}

   402→export interface DivergenceResult {

   409→/**
   410→ * مقایسه حرکت امروز نماد و DXY + وضعیت همبستگی ۳۰ روزه.
   411→ * برای طلا: رابطه معمول «طلا صعود ↔ دلار نزول» است.
   412→ * خروجی مشاهده است، نه اثبات علت.
   413→ */
   414→export function divergenceStatus(
   415→  symChangePct: number | null,
   416→  dxyChangePct: number | null,
   417→  corr: number | null
   418→): DivergenceResult {
   419→  if (symChangePct === null || dxyChangePct === null) {
   420→    return {
   421→      level: "unknown",
   422→      color: "gray",
   423→      text: "داده لحظه‌ای کامل نیست",
   424→      reason: "قیمت امروز نماد یا DXY در دسترس نیست — وضعیت واگرایی قابل ارزیابی نیست.",
   425→    };
   426→  }
   427→  const s = symChangePct > 0.05 ? 1 : symChangePct < -0.05 ? -1 : 0;
   428→  const d = dxyChangePct > 0.05 ? 1 : dxyChangePct < -0.05 ? -1 : 0;
   429→  const corrTxt = corr === null ? "—" : Math.round(corr * 100) / 100;
   430→  const brokenCorr = corr !== null && corr > 0.6;
   431→  if (s !== 0 && d !== 0 && s === d) {
   432→    return {
   433→      level: brokenCorr ? "broken" : "weak",
   434→      color: brokenCorr ? "red" : "amber",
   435→      text: brokenCorr ? "هشدار: همبستگی شکسته شده!" : "همبستگی ضعیف: احتیاط کنید",
   436→      reason: `نماد و DXY هم‌جهت حرکت کرده‌اند (نماد ${symChangePct > 0 ? "↑" : "↓"} و DXY ${
   437→        dxyChangePct > 0 ? "↑" : "↓"

   457→/* ---------- ویژگی ۵: قواعد مربی رفتاری ---------- */
   458→
   459→export const NUDGE_RULES = {
   460→  refreshWindowMs: 5 * 60 * 1000,
   461→  refreshMax: 10,
   462→  revengeLockMs: 15 * 60 * 1000,
   463→  maxLossesPerDay: 3,
   464→} as const;
   465→
   466→/** شمارش رفرش‌های پنجره ۵ دقیقه اخیر + پیام هشدار */
   467→export function refreshNudge(timestamps: number[], now = Date.now()): { count: number; warn: boolean; message: string } {
   468→  const recent = timestamps.filter((t) => now - t < NUDGE_RULES.refreshWindowMs);
   469→  if (recent.length > NUDGE_RULES.refreshMax) {
   470→    return {
   471→      count: recent.length,
   472→      warn: true,
   473→      message: "بازار پرنوسان است، صبر کنید — بروزرسانی پیاپی تصمیم را بهتر نمی‌کند.",
   474→    };
   475→  }
   476→  return { count: recent.length, warn: false, message: "" };
   477→}

   479→/** بررسی قفل Revenge Trading پس از ثبت ضرر */
   480→export function revengeLock(lastLossAtISO: string | null, now = Date.now()): { locked: boolean; minutesLeft: number } {
   481→  if (!lastLossAtISO) return { locked: false, minutesLeft: 0 };
   482→  const t = new Date(lastLossAtISO).getTime();
   483→  if (!isFinite(t)) return { locked: false, minutesLeft: 0 };
   484→  const left = Math.ceil((t + NUDGE_RULES.revengeLockMs - now) / 60000);
   485→  return left > 0 ? { locked: true, minutesLeft: left } : { locked: false, minutesLeft: 0 };
   486→}

   489→export function restSuggestion(todayResults: ("win" | "loss" | "be")[]): { suggest: boolean; message: string } {
   490→  const losses = todayResults.filter((r) => r === "loss").length;

   495→    };
   496→  }
   497→  return { suggest: false, message: "" };

   546→export interface VisitSnapshot {
   547→  atISO: string;
   548→  trends: Record<string, string>; // کلید "xau:1h" → جهت
   549→  score: number | null;
   550→  divergence: string | null;
   551→  highEventId: string | null;
   552→  price: number | null;

   554→
   555→export interface VisitChange {
   556→  kind: "trend" | "score" | "divergence" | "event" | "price";
   557→  text: string;
   558→}
   559→
   560→export function diffSinceLastVisit(prev: VisitSnapshot, cur: VisitSnapshot): VisitChange[] {
   561→  const out: VisitChange[] = [];
   562→  for (const key of Object.keys(cur.trends)) {
   563→    const before = prev.trends?.[key];
   564→    const after = cur.trends[key];
   565→    if (before && after && before !== after) {
   566→      const [sym, tf] = key.split(":");
   567→      const symFa = sym === "xau" ? "طلا" : "داوجونز";
   568→      const tfFa: Record<string, string> = {
   569→        "15m": "۱۵ دقیقه",
   570→        "1h": "۱ ساعته",
   571→        "4h": "۴ ساعته",
   572→        "1d": "روزانه",
   573→      };
   574→      const dirFa: Record<string, string> = { bullish: "صعودی", bearish: "نزولی", neutral: "خنثی" };
   575→      out.push({
   576→        kind: "trend",
   577→        text: `جهت ساختار ${symFa} در تایم‌فریم ${tfFa[tf] ?? tf} از «${dirFa[before] ?? before}» به «${dirFa[after] ?? after}» تغییر کرد.`,
   578→      });
   579→    }
   580→  }
   581→  if (prev.score !== null && cur.score !== null && Math.abs(cur.score - prev.score) >= 10) {
   582→    out.push({
   583→      kind: "score",
   584→      text: `امتیاز اطمینان از ${prev.score} به ${cur.score} تغییر کرد.`,
   585→    });
   586→  }
   587→  if (prev.divergence && cur.divergence && prev.divergence !== cur.divergence) {
   588→    out.push({
   589→      kind: "divergence",
   590→      text: `وضعیت همبستگی با DXY از «${prev.divergence}» به «${cur.divergence}» تغییر کرد.`,
   591→    });
   592→  }
   593→  if (prev.highEventId && cur.highEventId && prev.highEventId !== cur.highEventId) {
   594→    out.push({ kind: "event", text: "خبر مهم جدیدی به تقویم اضافه یا جایگزین خبر قبلی شد." });
   595→  }
   596→  if (prev.price !== null && cur.price !== null && prev.price > 0) {
   597→    const pct = ((cur.price - prev.price) / prev.price) * 100;
   598→    if (Math.abs(pct) >= 0.5) {
   599→      out.push({
   600→        kind: "price",
   601→        text: `قیمت نماد نسبت به آخرین بازدید ${pct > 0 ? "+" : ""}${Math.round(pct * 100) / 100}٪ تغییر کرده است.`,
   602→      });
   603→    }
   604→  }
   605→  return out;
   606→}
