   482→var priceState = { mode: 'snapshot', lastTry: null, trying: false, okAt: null };
   483→var PROXIES = [
   484→  function (u) { return 'https://api.allorigins.win/raw?url=' + encodeURIComponent(u); },
   485→  function (u) { return 'https://corsproxy.io/?url=' + encodeURIComponent(u); },

   488→function fetchJSON(url, timeoutMs) {
   489→  return new Promise(function (resolve, reject) {
   490→    var ctrl = new AbortController();
   491→    var t = setTimeout(function () { ctrl.abort(); reject(new Error('timeout')); }, timeoutMs || 9000);
   492→    fetch(url, { signal: ctrl.signal })
   493→      .then(function (r) { if (!r.ok) throw new Error('http ' + r.status); return r.json(); })
   494→      .then(function (j) { clearTimeout(t); resolve(j); })
   495→      .catch(function (err) { clearTimeout(t); reject(err); });
   496→  });
   497→}
   498→function fetchYahooPrice(sym) {
   499→  var u = 'https://query1.finance.yahoo.com/v8/finance/chart/' + encodeURIComponent(sym) + '?range=1d&interval=1m';
   500→  var i = 0;
   501→  function attempt() {
   502→    if (i >= PROXIES.length) return Promise.resolve(null);
   503→    var p = PROXIES[i++];
   504→    return fetchJSON(p(u), 9000).then(function (j) {
   505→      var r = j && j.chart && j.chart.result && j.chart.result[0];
   506→      if (!r || !r.meta || typeof r.meta.regularMarketPrice !== 'number') throw new Error('bad payload');
   507→      var meta = r.meta;
   508→      return { price: meta.regularMarketPrice, prev: (meta.chartPreviousClose != null ? meta.chartPreviousClose : meta.previousClose) || null };
   509→    }).catch(function () { return attempt(); });
   510→  }
   511→  return attempt();
   512→}
   513→function refreshPrices() {
   514→  if (priceState.trying) return Promise.resolve(false);
   515→  priceState.trying = true;
   516→  setPriceBusy(true);
   517→  return Promise.all([fetchYahooPrice('GC=F'), fetchYahooPrice('^DJI')])
   518→    .then(function (rs) {
   519→      var ax = rs[0], ad = rs[1], ok = false;
   520→      if (ax && ms.xau) {
   521→        ms.xau.price = ax.price;
   522→        if (ax.prev) { ms.xau.prevClose = ax.prev; ms.xau.changePct = (ax.price - ax.prev) / ax.prev * 100; }
   523→        ok = true;
   524→      }
   525→      if (ad && ms.dji) {
   526→        ms.dji.price = ad.price;
   527→        if (ad.prev) { ms.dji.prevClose = ad.prev; ms.dji.changePct = (ad.price - ad.prev) / ad.prev * 100; }
   528→        ok = true;
   529→      }
   530→      priceState.lastTry = new Date();
   531→      priceState.trying = false;
   532→      setPriceBusy(false);
   533→      if (ok) {
   534→        priceState.mode = 'live';
   535→        priceState.okAt = new Date();
   536→        updatePriceWidgets(true);
   537→        updateStatusUI();
   538→        if (S.app && S.app.id === 'structure' && S.appPhase === 'open') renderApp('structure', true);
   539→      } else {
   540→        priceState.mode = 'offline';
   541→        updateStatusUI();
   542→      }
   543→      return ok;
   544→    })
   545→    .catch(function () { priceState.trying = false; setPriceBusy(false); priceState.mode = 'offline'; updateStatusUI(); return false; });
   546→}
