   140→function atrSeries(bars: ScalpBar[], len: number): number[] {
   141→  const out: number[] = new Array(bars.length).fill(NaN);
   142→  if (bars.length < len + 1) return out;
   143→  const trs: number[] = [bars[0].h - bars[0].l];
   144→  for (let i = 1; i < bars.length; i++) {
   145→    trs.push(
   146→      Math.max(
   147→        bars[i].h - bars[i].l,
   148→        Math.abs(bars[i].h - bars[i - 1].c),
   149→        Math.abs(bars[i].l - bars[i - 1].c),
   150→      ),
   151→    );
   152→  }
   153→  let a = 0;
   154→  for (let i = 0; i < len; i++) a += trs[i];
   155→  out[len] = a / len;
   156→  for (let i = len + 1; i < bars.length; i++) {
   157→    a = (a * (len - 1) + trs[i]) / len;
   158→    out[i] = a;
   159→  }
   160→  return out;
   161→}

   163→function rma(values: number[], len: number): number[] {
   164→  const out: number[] = new Array(values.length).fill(NaN);
   165→  let sum = 0;
   166→  let prev = NaN;
   167→  for (let i = 0; i < values.length; i++) {
   168→    if (i < len) {
   169→      sum += values[i];
   170→      if (i === len - 1) {
   171→        prev = sum / len;
   172→        out[i] = prev;
   173→      }
   174→    } else {
   175→      prev = (prev * (len - 1) + values[i]) / len;
   176→      out[i] = prev;
   177→    }
   178→  }
   179→  return out;
   180→}

   182→function rsiSeries(c: number[], len = 14): number[] {
   183→  const ch: number[] = [0];
   184→  for (let i = 1; i < c.length; i++) ch.push(c[i] - c[i - 1]);
   185→  const gains = ch.map((d) => (d > 0 ? d : 0));
   186→  const losses = ch.map((d) => (d < 0 ? -d : 0));
   187→  const ag = rma(gains, len);
   188→  const al = rma(losses, len);
   189→  const out: number[] = c.map(() => NaN);
   190→  for (let i = 0; i < c.length; i++) {
   191→    if (isNaN(ag[i]) || isNaN(al[i])) continue;
   192→    out[i] = al[i] === 0 ? (ag[i] === 0 ? 50 : 100) : 100 - 100 / (1 + ag[i] / al[i]);
   193→  }
   194→  return out;
   195→}

   197→function macdHistSeries(c: number[], fast = 12, slow = 26, sig = 9): number[] {
   198→  const ema = (len: number): number[] => {
   199→    const k = 2 / (len + 1);
   200→    const out: number[] = [];
   201→    let e = c[0];
   202→    for (let i = 0; i < c.length; i++) {
   203→      e = i === 0 ? c[0] : c[i] * k + e * (1 - k);
   204→      out.push(e);
   205→    }
   206→    return out;
   207→  };
   208→  const f = ema(fast);
   209→  const s = ema(slow);
   210→  const line = f.map((v, i) => v - s[i]);
   211→  const k = 2 / (sig + 1);
   212→  let se = line[0];
   213→  return line.map((v, i) => {
   214→    se = i === 0 ? v : v * k + se * (1 - k);
   215→    return v - se;
   216→  });
   217→}

   219→function stochSeries(bars: ScalpBar[], len = 14): number[] {
   220→  const out: number[] = bars.map(() => NaN);
   221→  for (let i = len - 1; i < bars.length; i++) {
   222→    let hh = -Infinity;
   223→    let ll = Infinity;
   224→    for (let j = i - len + 1; j <= i; j++) {
   225→      if (bars[j].h > hh) hh = bars[j].h;
   226→      if (bars[j].l < ll) ll = bars[j].l;
   227→    }
   228→    out[i] = hh === ll ? 50 : ((bars[i].c - ll) / (hh - ll)) * 100;
   229→  }
   230→  return out;
   231→}

   233→function cciSeries(bars: ScalpBar[], len = 20): number[] {
   234→  const tp = bars.map((b) => (b.h + b.l + b.c) / 3);
   235→  const out: number[] = bars.map(() => NaN);
   236→  for (let i = len - 1; i < bars.length; i++) {
   237→    let mean = 0;
   238→    for (let j = i - len + 1; j <= i; j++) mean += tp[j];
   239→    mean /= len;
   240→    let dev = 0;
   241→    for (let j = i - len + 1; j <= i; j++) dev += Math.abs(tp[j] - mean);
   242→    dev /= len;
   243→    out[i] = dev === 0 ? 0 : (tp[i] - mean) / (0.015 * dev);
   244→  }
   245→  return out;
   246→}
   247→
   248→function mfiSeries(bars: ScalpBar[], len = 14): number[] {
   249→  const tp = bars.map((b) => (b.h + b.l + b.c) / 3);
   250→  const out: number[] = bars.map(() => NaN);
   251→  for (let i = len; i < bars.length; i++) {
   252→    let pos = 0;
   253→    let neg = 0;
   254→    for (let j = i - len + 1; j <= i; j++) {
   255→      const flow = tp[j] * bars[j].c;
   256→      if (tp[j] > tp[j - 1]) pos += flow;
   257→      else if (tp[j] < tp[j - 1]) neg += flow;
   258→    }
   259→    out[i] = neg === 0 ? 100 : pos === 0 ? 0 : 100 - 100 / (1 + pos / neg);
   260→  }
   261→  return out;
   262→}

   264→interface Pivot {
   265→  idx: number; // اندیس کندل پیوت
   266→  price: number;

   269→/** پیوت‌های تأییدشده — پیوت در کندل idx وقتی تأیید می‌شود که idx+right پردازش شود */
   270→function pivots(highs: number[], lows: number[], left: number, right: number) {
   271→  const highsP: Pivot[] = [];
   272→  const lowsP: Pivot[] = [];
   273→  for (let i = left; i < highs.length - right; i++) {
   274→    let isH = true;
   275→    let isL = true;
   276→    for (let k = 1; k <= left; k++) {
   277→      if (highs[i - k] >= highs[i]) isH = false;
   278→      if (lows[i - k] <= lows[i]) isL = false;
   279→    }
   280→    for (let k = 1; k <= right; k++) {
   281→      if (highs[i + k] >= highs[i]) isH = false;
   282→      if (lows[i + k] <= lows[i]) isL = false;
   283→    }
   284→    if (isH) highsP.push({ idx: i, price: highs[i] });
   285→    if (isL) lowsP.push({ idx: i, price: lows[i] });
   286→  }
   287→  return { highsP, lowsP };
   288→}

   290→function rp(v: number): number {
   291→  if (!isFinite(v)) return 0;
   292→  if (v >= 1000) return Math.round(v * 100) / 100;
   293→  if (v >= 10) return Math.round(v * 100) / 100;
   294→  return Math.round(v * 100000) / 100000;
   295→}

   299→export function runScalpEngine(
   300→  bars: ScalpBar[],
   301→  h4bars: ScalpBar[] | null,
   302→  tfLabel = "",
   303→): ScalpResult {
   304→  const notesFa: string[] = [];
   305→  const n = bars.length;
   306→  const high = bars.map((b) => b.h);
   307→  const low = bars.map((b) => b.l);
   308→  const close = bars.map((b) => b.c);
   309→  const open = bars.map((b) => b.o);
   310→  const atr = atrSeries(bars, ATR_LEN);
   311→
   312→  /* --- ۴ساعته: نگاشت bucket → سقف/کف کندلِ بسته‌شدهٔ قبلی --- */
   313→  const h4Map = new Map<number, { h: number; l: number }>();
   314→  if (h4bars && h4bars.length >= 2) {
   315→    for (const b of h4bars) {
   316→      const k = Math.floor(b.t / H4_SEC);
   317→      const ex = h4Map.get(k);
   318→      if (!ex) h4Map.set(k, { h: b.h, l: b.l });
   319→      else {
   320→        ex.h = Math.max(ex.h, b.h);
   321→        ex.l = Math.min(ex.l, b.l);
   322→      }
   323→    }
   324→  }
   325→  const h4Ready = h4Map.size >= 2;
   326→  if (!h4Ready) notesFa.push("دادهٔ کندل‌های ۴ساعته کافی نبود؛ ستاپ ۴ساعته محاسبه نشد و فقط موج/OB/واگرایی فعال‌اند.");
   327→
   328→  /* --- پیوت‌ها برای واگرایی و شکست ساختار --- */
   329→  const { highsP: divPivotHighs, lowsP: divPivotLows } = pivots(high, low, 5, 5);
   330→  const { highsP: stPivotHighs, lowsP: stPivotLows } = pivots(high, low, PIVOT_LEN, PIVOT_LEN);
   331→  let stLowPtr = 0;
   332→  let stHighPtr = 0;
   333→  let lastPivotLow: number | null = null;
   334→  let lastPivotHigh: number | null = null;
   335→
   336→  /* --- وضعیت سوئینگ/موج --- */
   337→  let swingHigh: number | null = null;
   338→  let swingLow: number | null = null;
   339→  let hiBroken = false;
   340→  let loBroken = false;
   341→  let swPtr = 0;
   342→  let slPtr = 0;
   343→
   344→  interface WaveState {
   345→    active: boolean;
   346→    brkLvl: number;
   347→    atrAtBrk: number;
   348→    extreme: number;
   349→    startIdx: number;
   350→  }
   351→  let bear: WaveState | null = null;
   352→  let bull: WaveState | null = null;
   353→
   354→  /* --- زون‌های OB --- */
   355→  const supply: ScalpZone[] = [];
   356→  const demand: ScalpZone[] = [];
   357→  const overlap = (arr: ScalpZone[], t: number, b: number) =>
   358→    arr.some((z) => z.live && z.top > b && t > z.bot);
   359→
   360→  /* --- ستاپ ۴ساعته --- */
   361→  let curBucket: number | null = null;
   362→  let sellArmed = false;
   363→  let buyArmed = false;
   364→  let sellWaiting = false;
   365→  let buyWaiting = false;
   366→  let sellZone: { lo: number; hi: number } | null = null;
   367→  let buyZone: { lo: number; hi: number } | null = null;
   368→  let h4HighNow: number | null = null;
   369→  let h4LowNow: number | null = null;
   370→
   371→  /* --- واگرایی: سری‌های ابزارها --- */
   372→  const rsi = rsiSeries(close, 14);
   373→  const macdH = macdHistSeries(close);
   374→  const stoch = stochSeries(bars, 14);
   375→  const cci = cciSeries(bars, 20);
   376→  const mfi = mfiSeries(bars, 14);
   377→
   378→  const signals: ScalpSignal[] = [];
   379→  let lastDiv: { side: "bull" | "bear"; votes: number; hidden: boolean; idx: number } | null = null;
   380→  let lastBullDivIdx: number | null = null;
   381→  let lastBearDivIdx: number | null = null;
   382→
   383→  const oscVotesAt = (idx: number, prevIdx: number, dir: "bull" | "bear" | "hiddenBull" | "hiddenBear") => {
   384→    const pairs: [number, number][] = [
   385→      [rsi[idx], rsi[prevIdx]],
   386→      [macdH[idx], macdH[prevIdx]],
   387→      [stoch[idx], stoch[prevIdx]],
   388→      [cci[idx], cci[prevIdx]],
   389→      [mfi[idx], mfi[prevIdx]],
   390→    ];
   391→    let votes = 0;
   392→    for (const [a, b] of pairs) {
   393→      if (isNaN(a) || isNaN(b)) continue;
   394→      if ((dir === "bull" && a > b) || (dir === "bear" && a < b) || (dir === "hiddenBull" && a < b) || (dir === "hiddenBear" && a > b)) votes++;
   395→    }
   396→    return votes;
   397→  };
   398→
   399→  /* ================= حلقهٔ اصلی کندل‌ها ================= */
   400→  for (let i = 0; i < n; i++) {
   401→    const bar = bars[i];
   402→    const atrVal = atr[i];
   403→
   404→    /* --- ۱) پیوت‌های سوئینگ (تأیید SWING_LEN کندل بعد) --- */
   405→    while (swPtr < divPivotHighs.length && divPivotHighs[swPtr].idx + SWING_LEN <= i) {
   406→      swingHigh = divPivotHighs[swPtr].price;
   407→      hiBroken = false;
   408→      swPtr++;
   409→    }
   410→    while (slPtr < divPivotLows.length && divPivotLows[slPtr].idx + SWING_LEN <= i) {
   411→      swingLow = divPivotLows[slPtr].price;
   412→      loBroken = false;
   413→      slPtr++;
   414→    }
   415→
   416→    /* --- ۲) پیوت‌های شکست ساختار (PIVOT_LEN) --- */
   417→    while (stHighPtr < stPivotHighs.length && stPivotHighs[stHighPtr].idx + PIVOT_LEN <= i) {
   418→      lastPivotHigh = stPivotHighs[stHighPtr].price;
   419→      stHighPtr++;
   420→    }
   421→    while (stLowPtr < stPivotLows.length && stPivotLows[stLowPtr].idx + PIVOT_LEN <= i) {
   422→      lastPivotLow = stPivotLows[stLowPtr].price;
   423→      stLowPtr++;
   424→    }
   425→
   426→    /* --- ۳) نگهداری زون‌های OB (لمس/تخفیف) — قبل از منطق شکست، مثل ترتیب Pine --- */
   427→    let tapSellZone: ScalpZone | null = null;
   428→    let tapBuyZone: ScalpZone | null = null;
   429→    for (const z of supply) {
   430→      if (!z.live) continue;
   431→      const wickIn = bar.h >= z.bot;
   432→      if (z.tappedIdx === null && wickIn) {
   433→        z.tappedIdx = i;
   434→        tapSellZone = z;
   435→      }
   436→      const mit = close[i] >= z.bot; // obMitRule = Close — مطابق Pine: بسته‌شدن بالای کفِ زون فروش
   437→      if (mit) {
   438→        z.live = false;
   439→        z.fadedIdx = i;
   440→      }
   441→    }
   442→    for (const z of demand) {
   443→      if (!z.live) continue;
   444→      const wickIn = bar.l <= z.top;
   445→      if (z.tappedIdx === null && wickIn) {
   446→        z.tappedIdx = i;
   447→        tapBuyZone = z;
   448→      }
   449→      const mit = close[i] <= z.top; // بسته‌شدن زیر سقفِ زون خرید
   450→      if (mit) {
   451→        z.live = false;
   452→        z.fadedIdx = i;
   453→      }
   454→    }
   455→    if (tapSellZone)
   456→      signals.push({ side: "sell", kind: "ob-tap", idx: i, t: bar.t, price: bar.c, zoneLo: tapSellZone.bot, zoneHi: tapSellZone.top, obConfluence: true, patternFa: "برخورد اول قیمت به زون فروش (OB)" });
   457→    if (tapBuyZone)
   458→      signals.push({ side: "buy", kind: "ob-tap", idx: i, t: bar.t, price: bar.c, zoneLo: tapBuyZone.bot, zoneHi: tapBuyZone.top, obConfluence: true, patternFa: "برخورد اول قیمت به زون خرید (OB)" });
   459→
   460→    /* --- ۴) ماشین حالت موج شکست --- */
   461→    let loBreakAny = false;
   462→    let hiBreakAny = false;
   463→    if (swingLow !== null && !isNaN(atrVal) && atrVal > 0) {
   464→      if (loBroken && close[i] > swingLow) loBroken = false;
   465→      if (close[i] < swingLow) {
   466→        if (!loBroken) {
   467→          loBreakAny = true;
   468→          if (!bear) bear = { active: true, brkLvl: swingLow, atrAtBrk: atrVal, extreme: bar.l, startIdx: i };
   469→        }
   470→        loBroken = true;
   471→      }
   472→    }
   473→    if (swingHigh !== null && !isNaN(atrVal) && atrVal > 0) {
   474→      if (hiBroken && close[i] < swingHigh) hiBroken = false;
   475→      if (close[i] > swingHigh) {
   476→        if (!hiBroken) {
   477→          hiBreakAny = true;
   478→          if (!bull) bull = { active: true, brkLvl: swingHigh, atrAtBrk: atrVal, extreme: bar.h, startIdx: i };
   479→        }
   480→        hiBroken = true;
   481→      }
   482→    }
   483→    if (bear && bear.active) {
   484→      if (bar.l < bear.extreme) bear.extreme = bar.l;
   485→      if (i > bear.startIdx) {
   486→        const rng = bear.brkLvl - bear.extreme;
   487→        const retr = bear.extreme + rng * (RETR_PCT / 100);
   488→        if (close[i] > retr) bear = null; // موج نزولی پایان یافت
   489→      }
   490→    }
   491→    if (bull && bull.active) {
   492→      if (bar.h > bull.extreme) bull.extreme = bar.h;
   493→      if (i > bull.startIdx) {
   494→        const rng = bull.extreme - bull.brkLvl;
   495→        const retr = bull.extreme - rng * (RETR_PCT / 100);
   496→        if (close[i] < retr) bull = null;
   497→      }
   498→    }
   499→
   500→    /* --- ۵) ساخت OB روی شکست تازهٔ سوئینگ --- */
   501→    if (loBreakAny) {
   502→      let oB: number | null = null;
   503→      for (let k = 1; k <= OB_LOOK && i - k >= 0; k++) {
   504→        if (close[i - k] > open[i - k]) {
   505→          oB = k;
   506→          break;
   507→        }
   508→      }
   509→      if (oB !== null) {
   510→        const zt = high[i - oB];
   511→        const zb = low[i - oB];
   512→        let fvg = !OB_FVG_REQ;
   513→        if (OB_FVG_REQ && oB >= 2) {
   514→          for (let a = 0; a <= oB - 2; a++) {
   515→            if (low[i - a - 2] > high[i - a]) {
   516→              fvg = true;
   517→              break;
   518→            }
   519→          }
   520→        }
   521→        let left = false;
   522→        let preMit = false;
   523→        for (let k2 = oB - 1; k2 >= 0; k2--) {
   524→          if (!left) {
   525→            if (close[i - k2] < zb) left = true;
   526→          } else if (close[i - k2] >= zb) {
   527→            preMit = true;
   528→            break;
   529→          }
   530→        }
   531→        if (fvg && left && !preMit && !overlap(supply, zt, zb)) {
   532→          supply.push({ side: "supply", top: zt, bot: zb, live: true, createdIdx: i, tappedIdx: null, fadedIdx: null });
   533→          if (supply.length > OB_MAX_ZONES) supply.shift();
   534→        }
   535→      }
   536→    }
   537→    if (hiBreakAny) {
   538→      let oU: number | null = null;
   539→      for (let k = 1; k <= OB_LOOK && i - k >= 0; k++) {
   540→        if (close[i - k] < open[i - k]) {
   541→          oU = k;
   542→          break;
   543→        }
   544→      }
   545→      if (oU !== null) {
   546→        const zt = high[i - oU];
   547→        const zb = low[i - oU];
   548→        let fvg = !OB_FVG_REQ;
   549→        if (OB_FVG_REQ && oU >= 2) {
   550→          for (let a = 0; a <= oU - 2; a++) {
   551→            if (high[i - a - 2] < low[i - a]) {
   552→              fvg = true;
   553→              break;
   554→            }
   555→          }
   556→        }
   557→        let left = false;
   558→        let preMit = false;
   559→        for (let k2 = oU - 1; k2 >= 0; k2--) {
   560→          if (!left) {
   561→            if (close[i - k2] > zt) left = true;
   562→          } else if (close[i - k2] <= zt) {
   563→            preMit = true;
   564→            break;
   565→          }
   566→        }
   567→        if (fvg && left && !preMit && !overlap(demand, zt, zb)) {
   568→          demand.push({ side: "demand", top: zt, bot: zb, live: true, createdIdx: i, tappedIdx: null, fadedIdx: null });
   569→          if (demand.length > OB_MAX_ZONES) demand.shift();
   570→        }
   571→      }
   572→    }
   573→
   574→    /* --- ۶) ستاپ ۴ساعته: برخورد ← شکست ← ریتست ← الگو --- */
   575→    const ocBuy = demand.some((z) => z.live && bar.l <= z.top && bar.h >= z.bot);
   576→    const ocSell = supply.some((z) => z.live && bar.l <= z.top && bar.h >= z.bot);
   577→
   578→    if (h4Ready) {
   579→      const bucket = Math.floor(bar.t / H4_SEC);
   580→      if (curBucket === null) curBucket = bucket;
   581→      if (bucket !== curBucket) {
   582→        // کندل ۴ساعتهٔ جدید → ریست وضعیت (مطابق new4h در Pine)
   583→        curBucket = bucket;
   584→        sellArmed = false;
   585→        buyArmed = false;
   586→        sellWaiting = false;
   587→        buyWaiting = false;
   588→        sellZone = null;
   589→        buyZone = null;
   590→      }
   591→      // آخرین کندل ۴ساعتهٔ «بسته‌شده» قبل از bucket فعلی
   592→      for (let b = bucket - 1; b >= bucket - 12; b--) {
   593→        const e = h4Map.get(b);
   594→        if (e) {
   595→          h4HighNow = e.h;
   596→          h4LowNow = e.l;
   597→          break;
   598→        }
   599→      }
   600→      /* مطابق Pine: وضعیت «مسلح» پیش از کندل فعلی ملاک شکست است (sellWasArmed) */
   601→      const sellWasArmed = sellArmed;
   602→      const buyWasArmed = buyArmed;
   603→      if (h4HighNow !== null && bar.h >= h4HighNow) {
   604→        sellArmed = true;
   605→        sellWaiting = false;
   606→        sellZone = null;
   607→      }
   608→      if (h4LowNow !== null && bar.l <= h4LowNow) {
   609→        buyArmed = true;
   610→        buyWaiting = false;
   611→        buyZone = null;
   612→      }
   613→      // شکست ساختار با Close زیر آخرین پیوت کف (پیش از ریتست)
   614→      if (sellWasArmed && !sellWaiting && lastPivotLow !== null && close[i] < lastPivotLow) {
   615→        sellZone = { lo: bar.l, hi: bar.h };
   616→        sellWaiting = true;
   617→        sellArmed = false;
   618→      }
   619→      if (buyWasArmed && !buyWaiting && lastPivotHigh !== null && close[i] > lastPivotHigh) {
   620→        buyZone = { lo: bar.l, hi: bar.h };
   621→        buyWaiting = true;
   622→        buyArmed = false;
   623→      }
   624→    }
   625→
   626→    /* --- ۷) الگوهای کندلی --- */
   627→    const rng = bar.h - bar.l;
   628→    const body = Math.abs(bar.c - bar.o);
   629→    const prevBar = i > 0 ? bars[i - 1] : null;
   630→    const isDojiPrev = prevBar
   631→      ? prevBar.h - prevBar.l > 0 &&
   632→        Math.abs(prevBar.c - prevBar.o) <= ((prevBar.h - prevBar.l) * DOJI_BODY_PCT) / 100
   633→      : false;
   634→    const bearEngulf = bar.c < bar.o && i > 0 && close[i - 1] > open[i - 1] && bar.o >= close[i - 1] && bar.c <= open[i - 1];
   635→    const bullEngulf = bar.c > bar.o && i > 0 && close[i - 1] < open[i - 1] && bar.o <= close[i - 1] && bar.c >= open[i - 1];
   636→    const isKeyBar = rng > 0 && body >= (rng * KEYBAR_BODY_PCT) / 100 && !isNaN(atrVal) && body >= atrVal * KEYBAR_ATR_MULT;
   637→    const bullKey = isKeyBar && bar.c > bar.o;
   638→    const bearKey = isKeyBar && bar.c < bar.o;
   639→    const bullPattern = bullEngulf || (isDojiPrev && bullKey); // دوجیِ کندل قبل + کی‌بار فعلی (مطابق Pine)
   640→    const bearPattern = bearEngulf || (isDojiPrev && bearKey);
   641→    const bullPatternFa = bullEngulf ? "انگالف صعودی" : "دوجی + کی‌بار صعودی";
   642→    const bearPatternFa = bearEngulf ? "انگالف نزولی" : "دوجی + کی‌بار نزولی";
   643→
   644→    /* --- ۸) ریتست + سیگنال ۴ساعته --- */
   645→    if (sellWaiting && sellZone && bar.l <= sellZone.hi && bar.h >= sellZone.lo) {
   646→      if (bearPattern) {
   647→        signals.push({ side: "sell", kind: "setup4h", idx: i, t: bar.t, price: bar.c, zoneLo: sellZone.lo, zoneHi: sellZone.hi, obConfluence: ocSell, patternFa: bearPatternFa + (ocSell ? " + هم‌پوشانی OB" : "") });
   648→        sellWaiting = false;
   649→      }
   650→    }
   651→    if (buyWaiting && buyZone && bar.l <= buyZone.hi && bar.h >= buyZone.lo) {
   652→      if (bullPattern) {
   653→        signals.push({ side: "buy", kind: "setup4h", idx: i, t: bar.t, price: bar.c, zoneLo: buyZone.lo, zoneHi: buyZone.hi, obConfluence: ocBuy, patternFa: bullPatternFa + (ocBuy ? " + هم‌پوشانی OB" : "") });
   654→        buyWaiting = false;
   655→      }
   656→    }
   657→
   658→    /* --- ۹) واگرایی (توافق ≥۳ از ۵ ابزار) --- */
   659→    for (const p of divPivotHighs) {
   660→      if (p.idx + 5 !== i) continue;
   661→      const prev = [...divPivotHighs].reverse().find((q) => q.idx < p.idx);
   662→      if (!prev) continue;
   663→      const gap = p.idx - prev.idx;
   664→      if (gap < 5 || gap > 60) continue;
   665→      if (high[p.idx] > high[prev.idx]) {
   666→        const votes = oscVotesAt(p.idx, prev.idx, "bear");
   667→        if (votes >= 3) {
   668→          lastDiv = { side: "bear", votes, hidden: false, idx: i };
   669→          lastBearDivIdx = i;
   670→        }
   671→      } else {
   672→        const votes = oscVotesAt(p.idx, prev.idx, "hiddenBear");
   673→        if (votes >= 3) {
   674→          lastDiv = { side: "bear", votes, hidden: true, idx: i };
   675→          lastBearDivIdx = i;
   676→        }
   677→      }
   678→    }
   679→    for (const p of divPivotLows) {
   680→      if (p.idx + 5 !== i) continue;
   681→      const prev = [...divPivotLows].reverse().find((q) => q.idx < p.idx);
   682→      if (!prev) continue;
   683→      const gap = p.idx - prev.idx;
   684→      if (gap < 5 || gap > 60) continue;
   685→      if (low[p.idx] < low[prev.idx]) {
   686→        const votes = oscVotesAt(p.idx, prev.idx, "bull");
   687→        if (votes >= 3) {
   688→          lastDiv = { side: "bull", votes, hidden: false, idx: i };
   689→          lastBullDivIdx = i;
   690→        }
   691→      } else {
   692→        const votes = oscVotesAt(p.idx, prev.idx, "hiddenBull");
   693→        if (votes >= 3) {
   694→          lastDiv = { side: "bull", votes, hidden: true, idx: i };
   695→          lastBullDivIdx = i;
   696→        }
   697→      }
   698→    }
   699→  }
   700→
   701→  /* ================= جمع‌بندی خروجی ================= */
   702→  const last = bars[n - 1];
   703→  const lastAtr = isNaN(atr[n - 1]) ? null : atr[n - 1];
   704→
   705→  /* گیت کیفیت داده: کندل‌های بدون بدنه (open=close) → موتور OB/الگو/موج روی این داده معنا ندارد */
   706→  const tail = bars.slice(-200);
   707→  const bodyless = tail.filter((b) => b.o === b.c).length / tail.length;
   708→  const degraded = bodyless > 0.6;
   709→  if (degraded) {
   710→    notesFa.push(
   711→      `کیفیت داده: ${Math.round(bodyless * 100)}٪ از ۲۰۰ کندل اخیر این تایم‌فریم از منبع بدون بدنه‌اند (open=close) — زون‌های OB، الگوهای کندلی و موج روی این داده قابل محاسبه نیستند؛ تایم‌فریم بزرگ‌تر (۵ یا ۱۵ دقیقه) را انتخاب کنید.`,
   712→    );
   713→  }
   714→  const lastDivIdx = lastDiv ? lastDiv.idx : null;
   715→  const divRecent = lastDivIdx !== null && n - 1 - lastDivIdx <= 10;
   716→
   717→  const liveSupply = supply.filter((z) => z.live && z.bot > last.c).sort((a, b) => a.bot - b.bot).slice(0, 3);
   718→  const liveDemand = demand.filter((z) => z.live && z.top < last.c).sort((a, b) => b.top - a.top).slice(0, 3);
   719→
   720→  let waveOut: ScalpResult["wave"] = { active: null, nAtr: null, retrLevel: null };
   721→  if (bear && bear.active) {
   722→    const rngB = bear.brkLvl - bear.extreme;
   723→    waveOut = { active: "bear", nAtr: bear.atrAtBrk > 0 ? rngB / bear.atrAtBrk : null, retrLevel: bear.extreme + rngB * (RETR_PCT / 100) };
   724→  } else if (bull && bull.active) {
   725→    const rngU = bull.extreme - bull.brkLvl;
   726→    waveOut = { active: "bull", nAtr: bull.atrAtBrk > 0 ? rngU / bull.atrAtBrk : null, retrLevel: bull.extreme - rngU * (RETR_PCT / 100) };
   727→  }
   728→
   729→  /* پیشنهاد حدودی بر اساس تازه‌ترین سیگنال — با سنجش وضعیت نسبت به قیمت فعلی */
   730→  let suggestion: ScalpSuggestion | null = null;
   731→  const fresh = signals.filter((s) => n - 1 - s.idx <= SCALP_FRESH_BARS);
   732→  const latest = fresh.length ? fresh[fresh.length - 1] : null;
   733→  if (latest && lastAtr && lastAtr > 0) {
   734→    const mid = (latest.zoneLo + latest.zoneHi) / 2;
   735→    const slBuf = 0.5 * lastAtr;
   736→    const risk = latest.side === "buy" ? mid - (latest.zoneLo - slBuf) : latest.zoneHi + slBuf - mid;
   737→    if (risk > 0) {
   738→      const pickTp = (dir: 1 | -1, minDist: number): number => {
   739→        // نزدیک‌ترین سطح مخالف حداقل به فاصلهٔ minDist
   740→        const cands: number[] = [];
   741→        if (dir === 1) {
   742→          for (const z of liveSupply) cands.push(z.bot, z.top);
   743→          if (swingHigh !== null) cands.push(swingHigh);
   744→          if (h4HighNow !== null) cands.push(h4HighNow);
   745→          const ok = cands.filter((v) => v >= mid + minDist).sort((a, b) => a - b);
   746→          return ok.length ? ok[0] : mid + 2 * minDist;
   747→        }
   748→        for (const z of liveDemand) cands.push(z.bot, z.top);
   749→        if (swingLow !== null) cands.push(swingLow);
   750→        if (h4LowNow !== null) cands.push(h4LowNow);
   751→        const ok = cands.filter((v) => v <= mid - minDist).sort((a, b) => b - a);
   752→        return ok.length ? ok[0] : mid - 2 * minDist;
   753→      };
   754→      const tp1 = pickTp(latest.side === "buy" ? 1 : -1, risk);
   755→      const tp2 = pickTp(latest.side === "buy" ? 1 : -1, 2 * risk);
   756→      const entryLo = latest.zoneLo;
   757→      const entryHi = latest.zoneHi;
   758→      const sl = latest.side === "buy" ? entryLo - slBuf : entryHi + slBuf;
   759→
   760→      /* وضعیت سیگنال نسبت به قیمتِ «فعلی» (نه لحظهٔ صدور) — پاسخ به «چرا قیمت دور از ناحیه است؟ */
   761→      const dist =
   762→        last.c >= entryLo && last.c <= entryHi
   763→          ? 0
   764→          : latest.side === "buy"
   765→            ? last.c < entryLo
   766→              ? entryLo - last.c
   767→              : last.c - entryHi
   768→            : last.c > entryHi
   769→              ? last.c - entryHi
   770→              : entryLo - last.c;
   771→      let status: SuggestionStatus = "active";
   772→      let statusFa = "فعال — قیمت در ناحیه یا نزدیک آن است";
   773→      if (latest.side === "buy" && last.c < sl) {
   774→        status = "invalidated";
   775→        statusFa = "ابطال شده — قیمت از حد ضرر عبور کرده است";
   776→      } else if (latest.side === "sell" && last.c > sl) {
   777→        status = "invalidated";
   778→        statusFa = "ابطال شده — قیمت از حد ضرر عبور کرده است";
   779→      } else if (latest.side === "buy" && last.c >= tp1) {
   780→        status = "tp1_reached";
   781→        statusFa = "هدف اول لمس شده — برای ورود جدید به تأیید تازه نیاز است";
   782→      } else if (latest.side === "sell" && last.c <= tp1) {
   783→        status = "tp1_reached";
   784→        statusFa = "هدف اول لمس شده — برای ورود جدید به تأیید تازه نیاز است";
   785→      } else if (dist > 1.2 * lastAtr) {
   786→        status = "far";
   787→        statusFa = "قیمت فعلی دور از ناحیه است — سیگنال مال لحظهٔ صدور بود؛ اجرا فقط با بازگشت قیمت به ناحیه";
   788→      }
   789→
   790→      suggestion = {
   791→        side: latest.side,
   792→        entryLo: rp(entryLo),
   793→        entryHi: rp(entryHi),
   794→        sl: rp(sl),
   795→        tp1: rp(tp1),
   796→        tp2: rp(tp2),
   797→        kind: latest.kind,
   798→        reasonFa:
   799→          (latest.kind === "setup4h"
   800→            ? `ستاپ ۴ساعته (${latest.patternFa}${latest.obConfluence ? " + هم‌پوشانی OB" : ""})`
   801→            : latest.patternFa) +
   802→          ` — سطوح و اهداف حدودی‌اند (ATR۱۴ تایم‌فریم: ${rp(lastAtr)})`,
   803→        signalBarsAgo: n - 1 - latest.idx,
   804→        status,
   805→        statusFa,
   806→        dist: rp(dist),
   807→        distAtr: lastAtr > 0 ? Math.round((dist / lastAtr) * 100) / 100 : null,
   808→      };
   809→    }
   810→  }
   811→
   812→  let divNote = "واگرایی تازه‌ای در ۱۰ کندل اخیر ثبت نشده است.";
   813→  if (lastDiv && divRecent) {
   814→    const sideFa = lastDiv.side === "bull" ? "صعودی" : "نزولی";
   815→    divNote = `واگرایی ${lastDiv.hidden ? "مخفی " : ""}${sideFa} با توافق ${lastDiv.votes}/۵ ابزار، ${n - 1 - lastDiv.idx} کندل پیش ثبت شده است.`;
   816→  }
   817→
   818→  return {
   819→    ok: true,
   820→    bars: n,
   821→    lastClose: last.c,
   822→    lastT: last.t,
   823→    atr: lastAtr,
   824→    atrPct: lastAtr ? (lastAtr / last.c) * 100 : null,
   825→    h4High: h4HighNow,
   826→    h4Low: h4LowNow,
   827→    swingHigh,
   828→    swingLow,
   829→    swingHighBroken: hiBroken,
   830→    swingLowBroken: loBroken,
   831→    setup: {
   832→      sell: sellWaiting ? "retest" : sellArmed ? "armed" : "idle",
   833→      buy: buyWaiting ? "retest" : buyArmed ? "armed" : "idle",
   834→      sellZone,
   835→      buyZone,
   836→      unavailable: !h4Ready,
   837→    },
   838→    wave: waveOut,
   839→    supplyZones: liveSupply.map((z) => ({ ...z, top: rp(z.top), bot: rp(z.bot) })),
   840→    demandZones: liveDemand.map((z) => ({ ...z, top: rp(z.top), bot: rp(z.bot) })),
   841→    signals: signals.slice(-6),
   842→    divergence: {
   843→      recentBull: divRecent && lastDiv?.side === "bull",
   844→      recentBear: divRecent && lastDiv?.side === "bear",
   845→      lastSide: lastDiv ? lastDiv.side : null,
   846→      lastVotes: lastDiv ? lastDiv.votes : null,
   847→      lastHidden: lastDiv ? lastDiv.hidden : false,
   848→      lastBarsAgo: lastDivIdx !== null ? n - 1 - lastDivIdx : null,
   849→      noteFa: divNote,
   850→    },
   851→    suggestion,
   852→    degraded,
   853→    notesFa,
   854→  };
   855→}
