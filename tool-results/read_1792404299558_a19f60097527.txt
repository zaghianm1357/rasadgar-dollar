    34→function scAtr(bars, len) {
    35→  var out = [], i;
    36→  for (i = 0; i < bars.length; i++) out.push(NaN);
    37→  if (bars.length < len + 1) return out;
    38→  var trs = [bars[0].h - bars[0].l];
    39→  for (i = 1; i < bars.length; i++)
    40→    trs.push(Math.max(bars[i].h - bars[i].l, Math.abs(bars[i].h - bars[i - 1].c), Math.abs(bars[i].l - bars[i - 1].c)));
    41→  var a = 0;
    42→  for (i = 0; i < len; i++) a += trs[i];
    43→  out[len] = a / len;
    44→  for (i = len + 1; i < bars.length; i++) { a = (a * (len - 1) + trs[i]) / len; out[i] = a; }
    45→  return out;
    46→}
    47→function scRma(v, len) {
    48→  var out = [], sum = 0, prev = NaN, i;
    49→  for (i = 0; i < v.length; i++) {
    50→    if (i < len) { sum += v[i]; if (i === len - 1) { prev = sum / len; out.push(prev); } else out.push(NaN); }
    51→    else { prev = (prev * (len - 1) + v[i]) / len; out.push(prev); }
    52→  }
    53→  return out;
    54→}
    55→function scRsi(c, len) {
    56→  len = len || 14;
    57→  var ch = [0], i;
    58→  for (i = 1; i < c.length; i++) ch.push(c[i] - c[i - 1]);
    59→  var g = ch.map(function (d) { return d > 0 ? d : 0; });
    60→  var l = ch.map(function (d) { return d < 0 ? -d : 0; });
    61→  var ag = scRma(g, len), al = scRma(l, len), out = [];
    62→  for (i = 0; i < c.length; i++) {
    63→    if (isNaN(ag[i]) || isNaN(al[i])) { out.push(NaN); continue; }
    64→    out.push(al[i] === 0 ? (ag[i] === 0 ? 50 : 100) : 100 - 100 / (1 + ag[i] / al[i]));
    65→  }
    66→  return out;
    67→}
    68→function scMacdHist(c) {
    69→  function ema(len) {
    70→    var k = 2 / (len + 1), e = c[0], out = [], i;
    71→    for (i = 0; i < c.length; i++) { e = i === 0 ? c[0] : c[i] * k + e * (1 - k); out.push(e); }
    72→    return out;
    73→  }
    74→  var f = ema(12), s = ema(26), line = [], i;
    75→  for (i = 0; i < c.length; i++) line.push(f[i] - s[i]);
    76→  var k = 2 / 10, se = line[0], hist = [];
    77→  for (i = 0; i < line.length; i++) { se = i === 0 ? line[0] : line[i] * k + se * (1 - k); hist.push(line[i] - se); }
    78→  return hist;
    79→}
    80→function scStoch(bars, len) {
    81→  len = len || 14;
    82→  var out = [], i, j;
    83→  for (i = 0; i < bars.length; i++) out.push(NaN);
    84→  for (i = len - 1; i < bars.length; i++) {
    85→    var hh = -Infinity, ll = Infinity;
    86→    for (j = i - len + 1; j <= i; j++) { if (bars[j].h > hh) hh = bars[j].h; if (bars[j].l < ll) ll = bars[j].l; }
    87→    out[i] = hh === ll ? 50 : ((bars[i].c - ll) / (hh - ll)) * 100;
    88→  }
    89→  return out;
    90→}
    91→function scCci(bars, len) {
    92→  len = len || 20;
    93→  var tp = bars.map(function (b) { return (b.h + b.l + b.c) / 3; }), out = [], i, j;
    94→  for (i = 0; i < bars.length; i++) out.push(NaN);
    95→  for (i = len - 1; i < bars.length; i++) {
    96→    var mean = 0;
    97→    for (j = i - len + 1; j <= i; j++) mean += tp[j];
    98→    mean /= len;
    99→    var dev = 0;
   100→    for (j = i - len + 1; j <= i; j++) dev += Math.abs(tp[j] - mean);
   101→    dev /= len;
   102→    out[i] = dev === 0 ? 0 : (tp[i] - mean) / (0.015 * dev);
   103→  }
   104→  return out;
   105→}
   106→function scMfi(bars, len) {
   107→  len = len || 14;
   108→  var tp = bars.map(function (b) { return (b.h + b.l + b.c) / 3; }), out = [], i, j;
   109→  for (i = 0; i < bars.length; i++) out.push(NaN);
   110→  for (i = len; i < bars.length; i++) {
   111→    var pos = 0, neg = 0;
   112→    for (j = i - len + 1; j <= i; j++) {
   113→      var flow = tp[j] * bars[j].c;
   114→      if (tp[j] > tp[j - 1]) pos += flow; else if (tp[j] < tp[j - 1]) neg += flow;
   115→    }
   116→    out[i] = neg === 0 ? 100 : pos === 0 ? 0 : 100 - 100 / (1 + pos / neg);
   117→  }
   118→  return out;
   119→}
   120→function scPivots(high, low, L, R) {
   121→  var hs = [], ls = [], i, k, okH, okL;
   122→  for (i = L; i < high.length - R; i++) {
   123→    okH = true; okL = true;
   124→    for (k = 1; k <= L; k++) { if (high[i - k] >= high[i]) okH = false; if (low[i - k] <= low[i]) okL = false; }
   125→    for (k = 1; k <= R; k++) { if (high[i + k] >= high[i]) okH = false; if (low[i + k] <= low[i]) okL = false; }
   126→    if (okH) hs.push({ idx: i, price: high[i] });
   127→    if (okL) ls.push({ idx: i, price: low[i] });
   128→  }
   129→  return { hs: hs, ls: ls };
   130→}

   324→      if (gapH < 5 || gapH > 60) continue;
   325→      if (high[p.idx] > high[prevH.idx]) { var vB = votesAt(p.idx, prevH.idx, 'bear'); if (vB >= 3) { lastDiv = { side: 'bear', votes: vB, hidden: false, idx: i }; lastBearIdx = i; } }
   326→      else { var vHB = votesAt(p.idx, prevH.idx, 'hBear'); if (vHB >= 3) { lastDiv = { side: 'bear', votes: vHB, hidden: true, idx: i }; lastBearIdx = i; } }
   327→    }
   328→    for (k = 0; k < pivDiv.ls.length; k++) {
   329→      var p2 = pivDiv.ls[k];
   330→      if (p2.idx + 5 !== i) continue;
   331→      var prevL = null;
   332→      for (j = pivDiv.ls.length - 1; j >= 0; j--) if (pivDiv.ls[j].idx < p2.idx) { prevL = pivDiv.ls[j]; break; }
   333→      if (!prevL) continue;
   334→      var gapL = p2.idx - prevL.idx;
   335→      if (gapL < 5 || gapL > 60) continue;
   336→      if (low[p2.idx] < low[prevL.idx]) { var vBl = votesAt(p2.idx, prevL.idx, 'bull'); if (vBl >= 3) { lastDiv = { side: 'bull', votes: vBl, hidden: false, idx: i }; lastBullIdx = i; } }
   337→      else { var vHl = votesAt(p2.idx, prevL.idx, 'hBull'); if (vHl >= 3) { lastDiv = { side: 'bull', votes: vHl, hidden: true, idx: i }; lastBullIdx = i; } }
   338→    }
   339→  }
   340→
   341→  /* جمع‌بندی */
   342→  var last = bars[n - 1], lastAtr = isNaN(atr[n - 1]) ? null : atr[n - 1];
   343→  var tail = bars.slice(-200), bodyless = 0;
   344→  for (i = 0; i < tail.length; i++) if (tail[i].o === tail[i].c) bodyless++;
   345→  var degraded = tail.length > 0 && bodyless / tail.length > 0.6;
   346→  if (degraded) notes.push('کیفیت داده: ' + faDigits(Math.round(bodyless / tail.length * 100)) + '٪ از ۲۰۰ کندل اخیر این تایم‌فریم از منبع بدون بدنه‌اند (open=close) — زون‌های OB، الگوهای کندلی و موج روی این داده قابل محاسبه نیستند؛ تایم‌فریم بزرگ‌تر (۵ یا ۱۵ دقیقه) را انتخاب کنید.');
   347→
   348→  var lastDivIdx = lastDiv ? lastDiv.idx : null;
   349→  var divRecent = lastDivIdx !== null && (n - 1 - lastDivIdx) <= 10;
   350→  var liveSupply = supply.filter(function (z) { return z.live && z.bot > last.c; }).sort(function (a, b) { return a.bot - b.bot; }).slice(0, 3);
   351→  var liveDemand = demand.filter(function (z) { return z.live && z.top < last.c; }).sort(function (a, b) { return b.top - a.top; }).slice(0, 3);
   352→
   353→  var waveOut = { active: null, nAtr: null, retrLevel: null };
   354→  if (bear && bear.active) { var rB = bear.brk - bear.ext; waveOut = { active: 'bear', nAtr: bear.atrBrk > 0 ? rB / bear.atrBrk : null, retrLevel: bear.ext + rB * (RETR / 100) }; }
   355→  else if (bull && bull.active) { var rU = bull.ext - bull.brk; waveOut = { active: 'bull', nAtr: bull.atrBrk > 0 ? rU / bull.atrBrk : null, retrLevel: bull.ext - rU * (RETR / 100) }; }
   356→
   357→  /* پیشنهاد حدودی */
   358→  var sg = null;
   359→  var fresh = signals.filter(function (s) { return (n - 1 - s.idx) <= SCALP_FRESH; });
   360→  var latest = fresh.length ? fresh[fresh.length - 1] : null;
   361→  if (latest && lastAtr && lastAtr > 0) {
   362→    var mid = (latest.zoneLo + latest.zoneHi) / 2, buf = 0.5 * lastAtr;
   363→    var risk = latest.side === 'buy' ? mid - (latest.zoneLo - buf) : (latest.zoneHi + buf) - mid;
   364→    if (risk > 0) {
   365→      function pickTp(dir, minDist) {
   366→        var cands = [];
   367→        if (dir === 1) {
   368→          for (var q = 0; q < liveSupply.length; q++) { cands.push(liveSupply[q].bot); cands.push(liveSupply[q].top); }
   369→          if (swingHigh !== null) cands.push(swingHigh);
   370→          if (h4HighNow !== null) cands.push(h4HighNow);
   371→          var okv = cands.filter(function (v) { return v >= mid + minDist; }).sort(function (a, b2) { return a - b2; });
   372→          return okv.length ? okv[0] : mid + 2 * minDist;
   373→        }
   374→        for (var q2 = 0; q2 < liveDemand.length; q2++) { cands.push(liveDemand[q2].bot); cands.push(liveDemand[q2].top); }
   375→        if (swingLow !== null) cands.push(swingLow);
   376→        if (h4LowNow !== null) cands.push(h4LowNow);
   377→        var okv2 = cands.filter(function (v) { return v <= mid - minDist; }).sort(function (a, b2) { return b2 - a; });
   378→        return okv2.length ? okv2[0] : mid - 2 * minDist;
   379→      }
   380→      var tp1 = pickTp(latest.side === 'buy' ? 1 : -1, risk);
   381→      var tp2 = pickTp(latest.side === 'buy' ? 1 : -1, 2 * risk);
   382→      sg = {
   383→        side: latest.side, entryLo: latest.zoneLo, entryHi: latest.zoneHi,
   384→        sl: latest.side === 'buy' ? latest.zoneLo - buf : latest.zoneHi + buf,
   385→        tp1: tp1, tp2: tp2, kind: latest.kind, barsAgo: n - 1 - latest.idx,
   386→        reason: (latest.kind === 'setup4h' ? 'ستاپ ۴ساعته (' + latest.pat + ')' : latest.pat) + ' — سطوح و اهداف حدودی‌اند (ATR۱۴ تایم‌فریم: ' + scPx(lastAtr) + ')'
   387→      };
   388→    }
   389→  }
   390→
   391→  var divNote = 'واگرایی تازه‌ای در ۱۰ کندل اخیر ثبت نشده است.';
   392→  if (lastDiv && divRecent)
   393→    divNote = 'واگرایی ' + (lastDiv.hidden ? 'مخفی ' : '') + (lastDiv.side === 'bull' ? 'صعودی' : 'نزولی') + ' با توافق ' + faDigits(lastDiv.votes) + '/۵ ابزار، ' + faDigits(n - 1 - lastDiv.idx) + ' کندل پیش ثبت شده است.';
   394→
   395→  return {
   396→    bars: n, lastClose: last.c, lastT: last.t,
   397→    atr: lastAtr, atrPct: lastAtr ? lastAtr / last.c * 100 : null,
   398→    h4High: h4HighNow, h4Low: h4LowNow,
   399→    swingHigh: swingHigh, swingLow: swingLow, swingHighBroken: hiBroken, swingLowBroken: loBroken,
   400→    setup: {
   401→      sell: sellWaiting ? 'retest' : sellArmed ? 'armed' : 'idle',
   402→      buy: buyWaiting ? 'retest' : buyArmed ? 'armed' : 'idle',
   403→      sellZone: sellZone, buyZone: buyZone, unavailable: !h4Ready
   404→    },
   405→    wave: waveOut,
   406→    supplyZones: liveSupply, demandZones: liveDemand,
   407→    signals: signals.slice(-6),
   408→    divergence: {
   409→      recentBull: !!(divRecent && lastDiv && lastDiv.side === 'bull'),
   410→      recentBear: !!(divRecent && lastDiv && lastDiv.side === 'bear'),
   411→      note: divNote
   412→    },
   413→    suggestion: sg, degraded: degraded, notes: notes
   414→  };
   415→}

   417→/* ---------- دریافت داده از Yahoo با پروکسی ---------- */
   418→function scFetchBars(yahooSym, interval, range) {
   419→  var u = 'https://query1.finance.yahoo.com/v8/finance/chart/' + encodeURIComponent(yahooSym) + '?interval=' + interval + '&range=' + range + '&includePrePost=false';
   420→  var i = 0;
   421→  function attempt() {
   422→    if (i >= PROXIES.length) return Promise.reject(new Error('proxy-fail'));
   423→    var p = PROXIES[i++];
   424→    return fetchJSON(p(u), 12000).then(function (j) {
   425→      var r = j && j.chart && j.chart.result && j.chart.result[0];
   426→      var q = r && r.indicators && r.indicators.quote && r.indicators.quote[0];
   427→      if (!r || !q || !q.close || !r.timestamp) throw new Error('bad');
   428→      var out = [], k;
   429→      for (k = 0; k < q.close.length; k++) {
   430→        if (q.close[k] == null || q.open[k] == null || q.high[k] == null || q.low[k] == null) continue;
   431→        out.push({ t: r.timestamp[k], o: q.open[k], h: q.high[k], l: q.low[k], c: q.close[k] });
   432→      }
   433→      if (out.length < 80) throw new Error('thin');
   434→      return out;
   435→    }).catch(function () { return attempt(); });
   436→  }
   437→  return attempt();
   438→}
   439→function scAgg4h(bars) {
   440→  var H = 14400, map = {}, keys = [], i;
   441→  for (i = 0; i < bars.length; i++) {
   442→    var b = Math.floor(bars[i].t / H) * H;
   443→    if (map[b] === undefined) { map[b] = { t: b, o: bars[i].o, h: bars[i].h, l: bars[i].l, c: bars[i].c }; keys.push(b); }
   444→    else { var e = map[b]; e.h = Math.max(e.h, bars[i].h); e.l = Math.min(e.l, bars[i].l); e.c = bars[i].c; }
   445→  }
   446→  keys.sort(function (a, b) { return a - b; });
   447→  return keys.map(function (k) { return map[k]; });
   448→}

   451→function scKey() { return S_SCALP.sym + '@' + S_SCALP.tf; }
   452→function scalpCompute(force) {
   453→  var key = scKey();
   454→  var cached = SCALP_CACHE[key];
   455→  if (!force && cached && Date.now() - cached.at < 60000) return Promise.resolve(cached.payload);
   456→  var symCfg = SCALP_SYMS[S_SCALP.sym], tfCfg = SCALP_TFS[S_SCALP.tf];
   457→  if (!symCfg || !tfCfg) return Promise.reject(new Error('cfg'));
   458→  SCALP_BUSY = true;
   459→  var h4p = null;
   460→  return scFetchBars(symCfg.y, tfCfg.i, tfCfg.r).then(function (bars) {
   461→    return scFetchBars(symCfg.y, '1h', '3mo').then(function (h1) { h4p = scAgg4h(h1); return bars; }).catch(function () { return bars; });
   462→  }).then(function (bars) {
   463→    var engine = scalpRun(bars, h4p);
   464→    var rw = null;
   465→    try {
   466→      rw = trRangeComp(bars.map(function (b) { return b.h; }), bars.map(function (b) { return b.l; }), bars.map(function (b) { return b.c; }));
   467→    } catch (e) { rw = null; }
   468→    var payload = {
   469→      ok: true, sym: S_SCALP.sym, symbolLabel: symCfg.label, symbolMono: symCfg.mono,
   470→      tf: S_SCALP.tf, tfLabel: tfCfg.label, engine: engine, rangeWarn: rw, at: Date.now()
   471→    };
   472→    SCALP_CACHE[key] = { at: Date.now(), payload: payload };
   473→    SCALP_BUSY = false;
   474→    return payload;
   475→  }).catch(function (err) {
   476→    SCALP_BUSY = false;
   477→    throw err;
   478→  });
   479→}
