    87→  try {
    88→    navigator.vibrate?.(12);
    89→  } catch {
    90→    /* پشتیبانی نمی‌شود */
    91→  }

   104→
   105→function Shell() {
   106→  const { ms, msRefreshing, refreshMarket, now } = useShell();
   107→
   108→  /* حالت لرزش و جابه‌جایی آیکون‌ها */
   109→  const [wiggle, setWiggle] = useState(false);
   110→  const [gridOrder, setGridOrder] = useState<AppId[]>(DEFAULT_GRID_ORDER);
   111→  const [drag, setDrag] = useState<{ id: AppId; x: number; y: number } | null>(null);
   112→
   113→  /* مراکز اعلان و کنترل */
   114→  const [ncOpen, setNcOpen] = useState(false);
   115→  const [ccOpen, setCcOpen] = useState(false);
   116→  const [ncProgress, setNcProgress] = useState(0);
   117→  const [ccProgress, setCcProgress] = useState(0);
   118→  const ncProgRef = useRef(0);
   119→  const ccProgRef = useRef(0);
   120→
   121→  /* منوی جانبی */
   122→  const [drawerOpen, setDrawerOpen] = useState(false);
   123→
   124→  const setNc = useCallback((v: number) => {
   125→    ncProgRef.current = v;
   126→    setNcProgress(v);
   127→  }, []);
   128→  const setCc = useCallback((v: number) => {
   129→    ccProgRef.current = v;
   130→    setCcProgress(v);
   131→  }, []);
   132→
   133→  /* اجرای اپ با انیمیشن بزرگ‌نمایی از موقعیت آیکون (FLIP) */
   134→  const [app, setApp] = useState<{ id: AppId; rect: AppRect } | null>(null);
   135→  const [appPhase, setAppPhase] = useState<"opening" | "open" | "closing">("open");
   136→
   137→  /* ---------- لایه‌های تاریخچه — دکمه برگشت خود گوشی ---------- */
   138→  const lastPushedRef = useRef<string | null>(null);
   139→  const pushLayer = useCallback((name: string) => {
   140→    lastPushedRef.current = name;
   141→    try {
   142→      history.pushState({ sd: name }, "");
   143→    } catch {
   144→      lastPushedRef.current = null;
   145→    }
   146→  }, []);
   147→
   148→  const closeApp = useCallback(() => {
   149→    setAppPhase("closing");
   150→    setTimeout(() => setApp(null), 420);
   151→  }, []);
   152→
   153→  const closeLayerUI = useCallback(
   154→    (name: "app" | "nc" | "cc" | "drawer") => {
   155→      if (lastPushedRef.current === name) {
   156→        lastPushedRef.current = null;
   157→        try {
   158→          history.back();
   159→          return;
   160→        } catch {
   161→          /* fallback زیر */
   162→        }
   163→      }
   164→      if (name === "drawer") setDrawerOpen(false);
   165→      else if (name === "nc") {
   166→        setNcOpen(false);
   167→        setNc(0);
   168→      } else if (name === "cc") {
   169→        setCcOpen(false);
   170→        setCc(0);
   171→      } else closeApp();
   172→    },
   173→    [closeApp, setNc, setCc],
   174→  );
   175→
   176→  /* وضعیت جاری برای هندلر popstate (بدون stale closure) */
   177→  const stateRef = useRef({
   178→    drawerOpen: false,
   179→    ncOpen: false,
   180→    ccOpen: false,
   181→    app: null as null | { id: AppId; rect: AppRect },
   182→    appPhase: "open" as "opening" | "open" | "closing",
   183→  });
   184→  useEffect(() => {
   185→    stateRef.current = { drawerOpen, ncOpen, ccOpen, app, appPhase };
   186→  }, [drawerOpen, ncOpen, ccOpen, app, appPhase]);
   187→
   188→  useEffect(() => {
   189→    try {
   190→      if (history.state && (history.state as { sd?: string }).sd) history.replaceState(null, "");
   191→    } catch {
   192→      /* بی‌اهمیت */
   193→    }
   194→    const onPop = () => {
   195→      lastPushedRef.current = null;
   196→      const s = stateRef.current;
   197→      if (s.drawerOpen) setDrawerOpen(false);
   198→      else if (s.ncOpen) {
   199→        setNcOpen(false);
   200→        setNc(0);
   201→      } else if (s.ccOpen) {
   202→        setCcOpen(false);
   203→        setCc(0);
   204→      } else if (s.app && s.appPhase !== "closing") closeApp();
   205→    };
   206→    window.addEventListener("popstate", onPop);
   207→    return () => window.removeEventListener("popstate", onPop);
   208→  }, [closeApp, setNc, setCc]);
   209→
   210→  /* صفحات و پارالاکس */
   211→  const pagesRef = useRef<HTMLDivElement | null>(null);
   212→  const wallRef = useRef<HTMLDivElement | null>(null);
   213→  const deviceRef = useRef<HTMLDivElement | null>(null);
   214→  const [page, setPage] = useState(0);
   215→
   216→  const onPagesScroll = useCallback(() => {
   217→    const el = pagesRef.current;
   218→    if (!el) return;
   219→    const maxAbs = el.scrollWidth - el.clientWidth;
   220→    const p = maxAbs > 0 ? Math.min(1, Math.abs(el.scrollLeft) / maxAbs) : 0;
   221→    if (wallRef.current) {
   222→      wallRef.current.style.transform = `translateX(${(0.5 - p) * 44}px) scale(1.08)`;
   223→    }
   224→    setPage(Math.round(p * (PAGES - 1)));
   225→  }, []);
   226→
   227→  const jumpToPage = (i: number) => {
   228→    const el = pagesRef.current;
   229→    if (!el) return;
   230→    const maxAbs = el.scrollWidth - el.clientWidth;
   231→    el.scrollTo({ left: i === 0 ? 0 : -maxAbs, behavior: "smooth" });
   232→  };
   233→
   234→  /* حرکات کشیدن از بالا/پایین */
   235→  const gesture = useRef<{
   236→    type: "nc" | "cc" | "close-app";
   237→    startY: number;
   238→    id: number;
   239→  } | null>(null);
   240→
   241→  const onRootPointerDown = (e: React.PointerEvent) => {
   242→    if (wiggle || drag || gesture.current || drawerOpen || ncOpen || ccOpen) return;
   243→    const rect = deviceRef.current?.getBoundingClientRect();
   244→    if (!rect) return;
   245→    const y = e.clientY - rect.top;
   246→    const inApp = !!app && appPhase !== "closing";
   247→    if (y < 46 || y > rect.height - 44) {
   248→      /* لمس دکمه‌ها در نوار وضعیت/نوار خانه → کلیک مستقیم، نه حرکت */
   249→      const t = e.target as HTMLElement | null;
   250→      if (t?.closest?.("button,[role=button],a,select,input,textarea")) return;
   251→    }
   252→    if (y < 46) {
   253→      gesture.current = { type: "nc", startY: e.clientY, id: e.pointerId };
   254→      try {
   255→        deviceRef.current?.setPointerCapture(e.pointerId);
   256→      } catch {
   257→        /* بی‌اهمیت */
   258→      }
   259→    } else if (y > rect.height - 44) {
   260→      gesture.current = { type: inApp ? "close-app" : "cc", startY: e.clientY, id: e.pointerId };
   261→      try {
   262→        deviceRef.current?.setPointerCapture(e.pointerId);
   263→      } catch {
   264→        /* بی‌اهمیت */
   265→      }
   266→    }
   267→  };
   268→
   269→  const onRootPointerMove = (e: React.PointerEvent) => {
   270→    const g = gesture.current;
   271→    if (!g || e.pointerId !== g.id) return;
   272→    const dy = e.clientY - g.startY;
   273→    if (g.type === "nc") setNc(Math.max(0, Math.min(1, dy / 150)));
   274→    else if (g.type === "cc") setCc(Math.max(0, Math.min(1, -dy / 170)));
   275→  };
   276→
   277→  const onRootPointerUp = (e: React.PointerEvent) => {
   278→    const g = gesture.current;
   279→    gesture.current = null;
   280→    if (!g || e.pointerId !== g.id) return;
   281→    if (g.type === "nc") {
   282→      const open = ncProgRef.current > 0.4;
   283→      if (open) {
   284→        setNcOpen(true);
   285→        setNc(1);
   286→        pushLayer("nc");
   287→      } else setNc(0);
   288→    } else if (g.type === "cc") {
   289→      const open = ccProgRef.current > 0.4;
   290→      if (open) {
   291→        setCcOpen(true);
   292→        setCc(1);
   293→        pushLayer("cc");
   294→      } else setCc(0);
   295→    } else {
   296→      // کشیدن به بالا روی نوار خانه هنگام باز بودن اپ → بستن اپ
   297→      if (e.clientY - g.startY < -36 && app && appPhase !== "closing") closeLayerUI("app");
   298→    }
   299→  };
   300→
   301→  const openNC = () => {
   302→    setNcOpen(true);
   303→    setNc(1);
   304→    pushLayer("nc");
   305→  };
   306→  const openCC = () => {
   307→    setCcOpen(true);
   308→    setCc(1);
   309→    pushLayer("cc");
   310→  };
   311→  const openDrawer = () => {
   312→    if (drawerOpen) return;
   313→    if (ncOpen) closeLayerUI("nc");
   314→    if (ccOpen) closeLayerUI("cc");
   315→    setDrawerOpen(true);
   316→    pushLayer("drawer");
   317→  };
   318→  const navFromDrawer = (kind: "app" | "nc" | "cc", id?: AppId) => {
   319→    setDrawerOpen(false);
   320→    try {
   321→      history.replaceState({ sd: kind }, "");
   322→      lastPushedRef.current = kind;
   323→    } catch {
   324→      /* بی‌اهمیت */
   325→    }
   326→    buzz();
   327→    if (kind === "app" && id) launchApp(id, true);
   328→    else if (kind === "nc") {
   329→      setNcOpen(true);
   330→      setNc(1);
   331→    } else {
   332→      setCcOpen(true);
   333→      setCc(1);
   334→    }
   335→  };
   336→
   337→  /* ترتیب ذخیره‌شده آیکون‌ها */
   338→  useEffect(() => {
   339→    let alive = true;
   340→    (async () => {
   341→      try {
   342→        const raw = localStorage.getItem(GRID_ORDER_KEY);
   343→        if (raw) {
   344→          const arr = JSON.parse(raw) as AppId[];
   345→          if (Array.isArray(arr)) {
   346→            /* ادغام ترتیب ذخیره‌شده با اپ‌های جدید (اپ جدید به انتها اضافه می‌شود) */
   347→            const known = arr.filter((a) => a in APPS);
   348→            const merged = [...known, ...DEFAULT_GRID_ORDER.filter((d) => !known.includes(d))];
   349→            if (merged.length && alive) setGridOrder(merged);
   350→          }
   351→        }
   352→      } catch {
   353→        /* ترتیب پیش‌فرض */
   354→      }
   355→    })();
   356→    return () => {
   357→      alive = false;
   358→    };
   359→  }, []);
   360→
   361→  const persistOrder = (order: AppId[]) => {
   362→    try {
   363→      localStorage.setItem(GRID_ORDER_KEY, JSON.stringify(order));
   364→    } catch {
   365→      /* حالت ناشناس */
   366→    }
   367→  };
   368→
   369→  /* لانگ‌پرس → لرزش | در حالت لرزش → کشیدن */
   370→  const pressRef = useRef<{
   371→    id: AppId;
   372→    startX: number;
   373→    startY: number;
   374→    timer: ReturnType<typeof setTimeout> | null;
   375→    longFired: boolean;
   376→  } | null>(null);
   377→  const iconRefs = useRef<Partial<Record<AppId, HTMLButtonElement | null>>>({});
   378→
   379→  const cancelLongPress = () => {
   380→    if (pressRef.current?.timer) clearTimeout(pressRef.current.timer);
   381→  };
   382→
   383→  const startLongPress = (id: AppId, e: React.PointerEvent) => {
   384→    const el = e.currentTarget as HTMLElement;
   385→    try {
   386→      el.setPointerCapture(e.pointerId);
   387→    } catch {
   388→      /* بی‌اهمیت */
   389→    }
   390→    pressRef.current = {
   391→      id,
   392→      startX: e.clientX,
   393→      startY: e.clientY,
   394→      timer: setTimeout(() => {
   395→        setWiggle(true);
   396→        buzz();
   397→        if (pressRef.current) pressRef.current.longFired = true;
   398→      }, LONG_PRESS_MS),
   399→      longFired: false,
   400→    };
   401→  };
   402→
   403→  const startDrag = (id: AppId, e: React.PointerEvent) => {
   404→    const el = e.currentTarget as HTMLElement;
   405→    try {
   406→      el.setPointerCapture(e.pointerId);
   407→    } catch {
   408→      /* بی‌اهمیت */
   409→    }
   410→    setDrag({ id, x: e.clientX, y: e.clientY });
   411→  };
   412→
   413→  const onIconPointerMove = (id: AppId, e: React.PointerEvent) => {
   414→    if (wiggle) {
   415→      if (!drag || drag.id !== id) return;
   416→      setDrag({ id, x: e.clientX, y: e.clientY });
   417→      const target = gridOrder.findIndex((a) => {
   418→        const el = iconRefs.current[a];
   419→        if (!el) return false;
   420→        const r = el.getBoundingClientRect();
   421→        return e.clientX >= r.left && e.clientX <= r.right;
   422→      });
   423→      if (target >= 0 && target !== gridOrder.indexOf(id)) {
   424→        setGridOrder((prev) => {
   425→          const from = prev.indexOf(id);
   426→          const next = [...prev];
   427→          next.splice(from, 1);
   428→          next.splice(target, 0, id);
   429→          return next;
   430→        });
   431→      }
   432→      return;
   433→    }
   434→    const p = pressRef.current;
   435→    if (p && p.id === id) {
   436→      const dist = Math.hypot(e.clientX - p.startX, e.clientY - p.startY);
   437→      if (dist > 10) cancelLongPress();
   438→    }
   439→  };
   440→
   441→  const onIconPointerUp = () => {
   442→    cancelLongPress();
   443→    if (drag) {
   444→      persistOrder(gridOrder);
   445→      setDrag(null);
   446→      buzz();
   447→    }
   448→  };
   449→
   450→  const launchApp = (id: AppId, noPush = false) => {
   451→    const el = iconRefs.current[id];
   452→    const dRect = deviceRef.current?.getBoundingClientRect();
   453→    if (!dRect) return;
   454→    const r = el?.getBoundingClientRect() ?? dRect;
   455→    buzz();
   456→    setApp({
   457→      id,
   458→      rect: {
   459→        x: r.left - dRect.left,
   460→        y: r.top - dRect.top,
   461→        w: r.width,
   462→        h: r.height,
   463→        dw: dRect.width,
   464→        dh: dRect.height,
   465→      },
   466→    });
   467→    setAppPhase("opening");
   468→    requestAnimationFrame(() => requestAnimationFrame(() => setAppPhase("open")));
   469→    if (!noPush) pushLayer("app");
   470→  };
   471→
   472→  const onIconClick = (id: AppId) => {
   473→    if (pressRef.current?.longFired) {
   474→      if (pressRef.current) pressRef.current.longFired = false;
   475→      return;
   476→    }
   477→    if (wiggle) return;
   478→    launchApp(id);
   479→  };
   480→
   481→  /* بستن لایه بالایی با Esc */
   482→  useEffect(() => {
   483→    const onKey = (e: KeyboardEvent) => {
   484→      if (e.key !== "Escape") return;
   485→      if (drawerOpen) closeLayerUI("drawer");
   486→      else if (ncOpen) closeLayerUI("nc");
   487→      else if (ccOpen) closeLayerUI("cc");
   488→      else if (app) closeLayerUI("app");
   489→    };
   490→    window.addEventListener("keydown", onKey);
   491→    return () => window.removeEventListener("keydown", onKey);
   492→  }, [drawerOpen, ncOpen, ccOpen, app, closeLayerUI]);
   493→
   494→  /* راهنمای اولیه */
   495→  const [hint, setHint] = useState(false);
   496→  useEffect(() => {
   497→    let alive = true;
   498→    (async () => {
   499→      await Promise.resolve();
   500→      if (!alive) return;
   501→      try {
   502→        if (!localStorage.getItem(SHELL_HINT_KEY)) {
   503→          if (alive) setHint(true);
   504→          localStorage.setItem(SHELL_HINT_KEY, "1");
   505→        }
   506→      } catch {
   507→        /* حالت ناشناس */
   508→      }
   509→    })();
   510→    return () => {
   511→      alive = false;
   512→    };
   513→  }, []);
   514→
   515→  /* ---------- محتوای اپ‌ها ---------- */
   516→
   517→  const appContent = (() => {
   518→    if (!app) return null;
   519→    if (app.id === "news") return <NewsApp />;
   520→    if (app.id === "about") return <AboutApp />;
   521→    if (app.id === "chart") return <ChartApp />;
   522→    if (app.id === "trader") return <TraderApp />;
   523→    if (app.id === "structure")
   524→      return (
   525→        <div className="p-4">
   526→          <MarketStructurePanel
   527→            data={ms}
   528→            refreshing={msRefreshing}
   529→            onExternalRefresh={() => refreshMarket(true)}
   530→          />
   531→        </div>
   532→      );
   533→    return (
   534→      <div className="p-4">
   535→        {ms ? (
   536→          <TradeSuggestion data={ms} />
   537→        ) : (
   538→          <div className="flex flex-col gap-2">
   539→            <Skeleton className="h-10 w-full" />
   540→            <Skeleton className="h-40 w-full" />
   541→          </div>
   542→        )}
   543→      </div>
   544→    );
   545→  })();
   546→
   547→  const sheetTransform =
   548→    app && appPhase !== "open"
   549→      ? `translate(${app.rect.x}px, ${app.rect.y}px) scale(${app.rect.w / app.rect.dw}, ${app.rect.h / app.rect.dh})`
   550→      : "none";
   551→
   552→  return (
   553→    <div
   554→      ref={deviceRef}
   555→      onPointerDown={onRootPointerDown}
   556→      onPointerMove={onRootPointerMove}
   557→      onPointerUp={onRootPointerUp}
   558→      onPointerCancel={() => {
   559→        gesture.current = null;
   560→        setNc(ncOpen ? 1 : 0);
