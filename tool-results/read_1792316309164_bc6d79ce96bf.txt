   471→var priceState = { mode: 'snapshot', lastTry: null, trying: false, okAt: null };
   472→var PROXIES = [
   473→  function (u) { return 'https://api.allorigins.win/raw?url=' + encodeURIComponent(u); },
   474→  function (u) { return 'https://corsproxy.io/?url=' + encodeURIComponent(u); },

   477→function fetchJSON(url, timeoutMs) {
   478→  return new Promise(function (resolve, reject) {
   479→    var ctrl = new AbortController();
   480→    var t = setTimeout(function () { ctrl.abort(); reject(new Error('timeout')); }, timeoutMs || 9000);
   481→    fetch(url, { signal: ctrl.signal })
   482→      .then(function (r) { if (!r.ok) throw new Error('http ' + r.status); return r.json(); })
   483→      .then(function (j) { clearTimeout(t); resolve(j); })
   484→      .catch(function (err) { clearTimeout(t); reject(err); });
   485→  });
   486→}
   487→function fetchYahooPrice(sym) {
   488→  var u = 'https://query1.finance.yahoo.com/v8/finance/chart/' + encodeURIComponent(sym) + '?range=1d&interval=1m';
   489→  var i = 0;
   490→  function attempt() {
   491→    if (i >= PROXIES.length) return Promise.resolve(null);
   492→    var p = PROXIES[i++];
   493→    return fetchJSON(p(u), 9000).then(function (j) {
   494→      var r = j && j.chart && j.chart.result && j.chart.result[0];
   495→      if (!r || !r.meta || typeof r.meta.regularMarketPrice !== 'number') throw new Error('bad payload');
   496→      var meta = r.meta;
   497→      return { price: meta.regularMarketPrice, prev: (meta.chartPreviousClose != null ? meta.chartPreviousClose : meta.previousClose) || null };
   498→    }).catch(function () { return attempt(); });
   499→  }
   500→  return attempt();
   501→}
   502→function refreshPrices() {
   503→  if (priceState.trying) return Promise.resolve(false);
   504→  priceState.trying = true;
   505→  setPriceBusy(true);
   506→  return Promise.all([fetchYahooPrice('GC=F'), fetchYahooPrice('^DJI')])
   507→    .then(function (rs) {
   508→      var ax = rs[0], ad = rs[1], ok = false;
   509→      if (ax && ms.xau) {
   510→        ms.xau.price = ax.price;
   511→        if (ax.prev) { ms.xau.prevClose = ax.prev; ms.xau.changePct = (ax.price - ax.prev) / ax.prev * 100; }
   512→        ok = true;
   513→      }
   514→      if (ad && ms.dji) {
   515→        ms.dji.price = ad.price;
   516→        if (ad.prev) { ms.dji.prevClose = ad.prev; ms.dji.changePct = (ad.price - ad.prev) / ad.prev * 100; }
   517→        ok = true;
   518→      }
   519→      priceState.lastTry = new Date();
   520→      priceState.trying = false;
   521→      setPriceBusy(false);
   522→      if (ok) {
   523→        priceState.mode = 'live';
   524→        priceState.okAt = new Date();

   530→        updateStatusUI();

   534→    .catch(function () { priceState.trying = false; setPriceBusy(false); priceState.mode = 'offline'; updateStatusUI(); return false; });
   535→}
